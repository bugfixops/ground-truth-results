   272	    {
   273	        Disposable d = dialogViewModel.observeEvents()
   274	                .subscribe((event) -> {
   275	                    if (event.dialogTag == null || !event.dialogTag.equals(TAG_DELETE_TRACKERS_DIALOG) || deleteTrackersDialog == null)
   276	                        return;
   277	                    switch (event.type) {

   281	                            break;
   282	                        case NEGATIVE_BUTTON_CLICKED:
   283	                            deleteTrackersDialog.dismiss();
   284	                            break;
   285	                    }
   286	                });
   287	        disposables.add(d);
   288	    }

   374	    private void shareUrl()
   375	    {
   376	        MutableSelection<TrackerItem> selections = new MutableSelection<>();
   377	        selectionTracker.copySelection(selections);
   378	
   379	        disposables.add(Observable.fromIterable(selections)

    91	    @Override
    92	    public void submitList(@Nullable List<TrackerItem> list)
    93	    {
   152	        void bind(TrackerItem item)
   153	        {
   154	            Context context = itemView.getContext();
   155	
   156	            selectionKey = item;
   157	
   158	            TypedArray a = context.obtainStyledAttributes(new TypedValue().data, new int[] {
   159	                    R.attr.selectableColor,
   160	                    R.attr.defaultRectRipple,
   161	                    R.attr.colorError,
   162	                    R.attr.colorOk
   163	            });
   164	
   165	            if (isSelected)
   166	                Utils.setBackground(itemView, a.getDrawable(0));
   167	            else
   168	                Utils.setBackground(itemView, a.getDrawable(1));
   169	
   170	            binding.url.setText(item.url);
   171	
   172	            String status = "";
   173	            switch (item.status) {
   174	                case TrackerInfo.Status.NOT_CONTACTED:
   175	                    status = context.getString(R.string.tracker_state_not_contacted);
   176	                    break;
   177	                case TrackerInfo.Status.WORKING:
   178	                    status = context.getString(R.string.tracker_state_working);
   179	                    break;
   180	                case TrackerInfo.Status.UPDATING:
   180	                binding.trackerList,
   181	                new TrackerListAdapter.KeyProvider(adapter),
   182	                new TrackerListAdapter.ItemLookup(binding.trackerList),
   183	                StorageStrategy.createParcelableStorage(TrackerItem.class))
   184	                .withSelectionPredicate(SelectionPredicates.createSelectAnything())
   185	                .build();
   186	
   187	        selectionTracker.addObserver(new SelectionTracker.SelectionObserver<TrackerItem>() {
   188	            @Override
   189	            public void onSelectionChanged()
   190	            {
   191	                super.onSelectionChanged();
   192	
   193	                if (selectionTracker.hasSelection() && actionMode == null) {
   194	                    actionMode = activity.startSupportActionMode(actionModeCallback);
   195	                    setActionModeTitle(selectionTracker.getSelection().size());
   196	
   197	                } else if (!selectionTracker.hasSelection()) {
   198	                    if (actionMode != null)
   199	                        actionMode.finish();
   200	                    actionMode = null;
   201	
   202	                } else {
   203	                    setActionModeTitle(selectionTracker.getSelection().size());
   204	
   205	                    /* Show/hide menu items after change selected files */
   206	                    int size = selectionTracker.getSelection().size();
   207	                    if (size == 1 || size == 2)
   208	                        actionMode.invalidate();
   209	                }
   210	            }
   211	
   212	            @Override
   213	            public void onSelectionRestored()
   214	            {
   215	                super.onSelectionRestored();
   216	
   217	                actionMode = activity.startSupportActionMode(actionModeCallback);
   218	                setActionModeTitle(selectionTracker.getSelection().size());
   219	            }
   220	        });
   221	
   222	        if (savedInstanceState != null)
   223	            selectionTracker.onRestoreInstanceState(savedInstanceState);
   224	        adapter.setSelectionTracker(selectionTracker);
   225	
   226	        FragmentManager fm = getChildFragmentManager();
   227	        deleteTrackersDialog = (BaseAlertDialog)fm.findFragmentByTag(TAG_DELETE_TRACKERS_DIALOG);
   228	    }
