    20	package org.proninyaroslav.libretorrent.ui;
    21	
    22	import android.app.Dialog;
    23	import android.content.DialogInterface;
    24	import android.os.Bundle;
    25	import android.view.LayoutInflater;
    26	import android.view.View;
    27	import android.widget.Button;
    28	
    29	import androidx.annotation.NonNull;
    30	import androidx.annotation.Nullable;
    31	import androidx.appcompat.app.AlertDialog;
    32	import androidx.fragment.app.DialogFragment;
    33	import androidx.lifecycle.ViewModelProvider;
    34	
    35	import io.reactivex.Observable;
    36	import io.reactivex.subjects.PublishSubject;
    37	
    38	/*
    39	 * The base alert dialog fragment. Support customizing the layout, text, buttons, title and style.
    40	 * Don't dismiss when changing the device configuration, unlike the AlertDialog.
    41	 */
    42	
    43	public class BaseAlertDialog extends DialogFragment
    44	{
    45	    private static final String TAG = BaseAlertDialog.class.getSimpleName();
    46	
    47	    protected static final String TAG_TITLE = "title";
    48	    protected static final String TAG_MESSAGE = "message";
    49	    protected static final String TAG_POS_TEXT = "positive_test";
    50	    protected static final String TAG_NEG_TEXT = "negative_text";
    51	    protected static final String TAG_NEUTRAL_BUTTON = "neutral_button";
    52	    protected static final String TAG_RES_ID_VIEW = "res_id_view";
    53	    protected static final String TAG_AUTO_DISMISS = "auto_dismiss";
    54	    protected SharedViewModel viewModel;
    55	
    56	    public static class SharedViewModel extends androidx.lifecycle.ViewModel
    57	    {
    58	        private final PublishSubject<Event> dialogEvents = PublishSubject.create();
    59	
    60	        public Observable<Event> observeEvents()
    61	        {
    62	            return dialogEvents;
    63	        }
    64	
    65	        public void sendEvent(Event event)
    66	        {
    67	            dialogEvents.onNext(event);
    68	        }
    69	    }
    70	
    71	    public enum EventType
    72	    {
    73	        POSITIVE_BUTTON_CLICKED,
    74	        NEGATIVE_BUTTON_CLICKED,
    75	        NEUTRAL_BUTTON_CLICKED,
    76	        DIALOG_SHOWN
    77	    }
    78	
    79	    public static class Event
    80	    {
    81	        @Nullable
    82	        public String dialogTag;
    83	        public EventType type;
    84	
    85	        public Event(@Nullable String dialogTag, EventType type)
    86	        {
    87	            this.dialogTag = dialogTag;
    88	            this.type = type;
    89	        }
    90	    }
    91	
    92	    /* In the absence of any parameter need set 0 or null */
    93	
    94	    public static BaseAlertDialog newInstance(String title, String message, int resIdView,
    95	                                              String positiveText, String negativeText,
    96	                                              String neutralText, boolean autoDismiss)
    97	    {
    98	        BaseAlertDialog frag = new BaseAlertDialog();
    99	
   100	        Bundle args = new Bundle();
   101	        args.putString(TAG_TITLE, title);
   102	        args.putString(TAG_MESSAGE, message);
   103	        args.putString(TAG_POS_TEXT, positiveText);
   104	        args.putString(TAG_NEG_TEXT, negativeText);
   105	        args.putString(TAG_NEUTRAL_BUTTON, neutralText);
   106	        args.putInt(TAG_RES_ID_VIEW, resIdView);
   107	        args.putBoolean(TAG_AUTO_DISMISS, autoDismiss);
   108	
   109	        frag.setArguments(args);
   110	
   111	        return frag;
   112	    }
   113	
   114	    @NonNull
   115	    @Override
   116	    public Dialog onCreateDialog(Bundle savedInstanceState)
   117	    {
   118	        viewModel = new ViewModelProvider(getActivity()).get(SharedViewModel.class);
   119	
   120	        Bundle args = getArguments();
   121	        String title = args.getString(TAG_TITLE);
   122	        String message = args.getString(TAG_MESSAGE);
   123	        String positiveText = args.getString(TAG_POS_TEXT);
   124	        String negativeText = args.getString(TAG_NEG_TEXT);
   125	        String neutralText = args.getString(TAG_NEUTRAL_BUTTON);
   126	        int resIdView = args.getInt(TAG_RES_ID_VIEW);
   127	        boolean autoDismiss = args.getBoolean(TAG_AUTO_DISMISS);
   128	
   129	        LayoutInflater i = LayoutInflater.from(getActivity());
   130	        View v = null;
   131	        if (resIdView != 0)
   132	            v = i.inflate(resIdView, null);
   133	
   134	        return buildDialog(title, message, v, positiveText,
   135	                negativeText, neutralText, autoDismiss);
   136	    }
   137	
   138	    private Event makeEvent(EventType type)
   139	    {
   140	        return new Event(getTag(), type);
   141	    }
   142	
   143	    protected AlertDialog buildDialog(String title, String message,
   144	                                      View view, String positiveText,
   145	                                      String negativeText, String neutralText,
   146	                                      boolean autoDismiss)
   147	    {
   148	        AlertDialog.Builder dialog = new AlertDialog.Builder(getActivity());
   149	        if (title != null)
   150	            dialog.setTitle(title);
   151	
   152	        if (message != null)
   153	            dialog.setMessage(message);
   154	
   155	        if (view != null)
   156	            dialog.setView(view);
   157	
   158	        if (positiveText != null)
   159	            dialog.setPositiveButton(positiveText, null);
   160	
   161	        if (negativeText != null)
   162	            dialog.setNegativeButton(negativeText, null);
   163	
   164	        if (neutralText != null)
   165	            dialog.setNeutralButton(neutralText, null);
   166	
   167	        final AlertDialog alert = dialog.create();
   168	        alert.setOnShowListener((DialogInterface dialogInterface) -> {
   169	            Button positiveButton = alert.getButton(AlertDialog.BUTTON_POSITIVE);
   170	            Button negativeButton = alert.getButton(AlertDialog.BUTTON_NEGATIVE);
   171	            Button neutralButton = alert.getButton(AlertDialog.BUTTON_NEUTRAL);
   172	            if (positiveButton != null) {
   173	                positiveButton.setOnClickListener((v) -> {
   174	                    viewModel.sendEvent(makeEvent(EventType.POSITIVE_BUTTON_CLICKED));
   175	                    if (autoDismiss)
   176	                        dismiss();
   177	                });
   178	            }
   179	            if (negativeButton != null) {
   180	                negativeButton.setOnClickListener((v) -> {
   181	                    viewModel.sendEvent(makeEvent(EventType.NEGATIVE_BUTTON_CLICKED));
   182	                    if (autoDismiss)
   183	                        dismiss();
   184	                });
   185	            }
   186	            if (neutralButton != null) {
   187	                neutralButton.setOnClickListener((v) -> {
   188	                    viewModel.sendEvent(makeEvent(EventType.NEUTRAL_BUTTON_CLICKED));
   189	                    if (autoDismiss)
   190	                        dismiss();
   191	                });
   192	            }
   193	
   194	            viewModel.sendEvent(makeEvent(EventType.DIALOG_SHOWN));
   195	        });
   196	
   197	        return alert;
   198	    }
   199	}
