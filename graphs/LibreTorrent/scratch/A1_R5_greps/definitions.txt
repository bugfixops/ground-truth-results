    27	/*
    28	 * The class provides an abstract package model, sent from the service.
    29	 */
    30	
    31	public abstract class AbstractInfoParcel
    32	        implements Parcelable, Comparable
    33	{
    34	    public String parcelId;
    35	
    36	    protected AbstractInfoParcel()
    37	    {
    38	        parcelId = UUID.randomUUID().toString();
    39	    }
    40	
    41	    protected AbstractInfoParcel(String parcelId)
    42	    {
    43	        this.parcelId = parcelId;
    44	    }
    45	
    46	    protected AbstractInfoParcel(Parcel source)
    47	    {
    48	        parcelId = source.readString();
    49	    }
    50	
    51	    @Override
    52	    public void writeToParcel(Parcel dest, int flags)
    53	    {
    54	        dest.writeString(parcelId);
    55	    }
    56	
    57	    @Override
    58	    public abstract int hashCode();
    59	
    60	    @Override
    61	    public abstract boolean equals(Object o);
    62	}

    29	/*
    30	 * The class provides a package model with dynamically
    31	 * changing information about state of the torrent.
    32	 */
    33	
    34	public class AdvancedTorrentInfo extends AbstractInfoParcel
    35	{
    36	    public String torrentId = "";
    37	    public int totalSeeds = 0;
    38	    public int seeds = 0;
    39	    public int downloadedPieces = 0;
    40	    public long[] filesReceivedBytes = new long[0];
    41	    public double shareRatio = 0.0;
    42	    public long activeTime = 0L;
    43	    public long seedingTime = 0L;
    44	    public double availability = 0.0;
    45	    public double[] filesAvailability = new double[0];
    46	    public int leechers;
    47	    public int totalLeechers;
    48	
    49	    public AdvancedTorrentInfo()
    50	    {
    51	        super();
    52	    }
    53	
    54	    public AdvancedTorrentInfo(String torrentId, long[] filesReceivedBytes,
    55	                               int totalSeeds, int seeds, int downloadedPieces,
    56	                               double shareRatio, long activeTime, long seedingTime,
    57	                               double availability, double[] filesAvailability,
    58	                               int leechers, int totalLeechers)
    59	    {
    60	        super(torrentId);
    61	
    62	        this.torrentId = torrentId;
    63	        this.filesReceivedBytes = filesReceivedBytes;
    64	        this.totalSeeds = totalSeeds;
    65	        this.seeds = seeds;
    66	        this.downloadedPieces = downloadedPieces;
    67	        this.shareRatio = shareRatio;
    68	        this.activeTime = activeTime;
    69	        this.seedingTime = seedingTime;
    70	        this.availability = availability;
    71	        this.filesAvailability = filesAvailability;
    72	        this.leechers = leechers;
    73	        this.totalLeechers = totalLeechers;
    74	    }
    75	
    76	    public AdvancedTorrentInfo(Parcel source)
    77	    {
    78	        super(source);
    79	
    80	        torrentId = source.readString();
    81	        filesReceivedBytes = source.createLongArray();
    82	        totalSeeds = source.readInt();
    83	        seeds = source.readInt();
    84	        downloadedPieces = source.readInt();
    85	        shareRatio = source.readDouble();
    86	        activeTime = source.readLong();
    87	        seedingTime = source.readLong();
    88	        availability = source.readDouble();
    89	        filesAvailability = source.createDoubleArray();
    90	        leechers = source.readInt();
    91	        totalLeechers = source.readInt();
    92	    }
    93	
    94	    @Override
    95	    public int describeContents()
    96	    {
    97	        return 0;
    98	    }
    99	
   100	    @Override
   101	    public void writeToParcel(Parcel dest, int flags)
   102	    {
   103	        super.writeToParcel(dest, flags);
   104	
   105	        dest.writeString(torrentId);
   106	        dest.writeLongArray(filesReceivedBytes);

   126	                }

   130	                {
   131	                    return new AdvancedTorrentInfo[size];
   132	                }
   133	            };
   134	
   135	    @Override
   136	    public int compareTo(@NonNull Object another)
   137	    {
   138	        return torrentId.compareTo(((AdvancedTorrentInfo)another).torrentId);
   139	    }
   140	
   141	    @Override
   142	    public int hashCode()
   143	    {
   144	        int prime = 31, result = 1;
   145	
   146	        result = prime * result + ((torrentId == null) ? 0 : torrentId.hashCode());
   147	        result += Arrays.hashCode(filesReceivedBytes);
   148	        result = prime * result + totalSeeds;
   149	        result = prime * result + seeds;
   150	        result = prime * result + downloadedPieces;
   151	        long shareRationBits = Double.doubleToLongBits(shareRatio);
   152	        result = prime * result + (int) (shareRationBits ^ (shareRationBits >>> 32));
   153	        result = prime * result + (int) (activeTime ^ (activeTime >>> 32));
   154	        result = prime * result + (int) (seedingTime ^ (seedingTime >>> 32));
   155	        long availabilityBits = Double.doubleToLongBits(availability);
   156	        result = prime * result + (int) (availabilityBits ^ (availabilityBits >>> 32));
   157	        result += Arrays.hashCode(filesAvailability);
   158	        result = prime * result + leechers;
   159	        result = prime * result + totalLeechers;
   160	
   161	        return result;
   162	    }
   163	
   164	    @Override
   165	    public boolean equals(Object o)
   166	    {
   167	        if (!(o instanceof AdvancedTorrentInfo))
   168	            return false;
   169	
   170	        if (o == this)
   171	            return true;
   172	
   173	        AdvancedTorrentInfo state = (AdvancedTorrentInfo)o;
   174	
   175	        return (torrentId == null || torrentId.equals(state.torrentId)) &&
   176	                totalSeeds == state.totalSeeds &&
   177	                seeds == state.seeds &&
   178	                downloadedPieces == state.downloadedPieces &&
   179	                Arrays.equals(filesReceivedBytes, state.filesReceivedBytes) &&
   180	                shareRatio == state.shareRatio &&
   181	                activeTime == state.activeTime &&
   182	                seedingTime == state.seedingTime &&
   183	                availability == state.availability &&
   184	                Arrays.equals(filesAvailability, state.filesAvailability) &&
   185	                leechers == state.leechers &&
   186	                totalLeechers == state.totalLeechers;
   187	    }
   188	
   189	    @NonNull
   190	    @Override
   191	    public String toString()
   192	    {
   193	        return "AdvancedTorrentInfo{" +
   194	                "torrentId='" + torrentId + '\'' +
   195	                ", totalSeeds=" + totalSeeds +
   196	                ", seeds=" + seeds +
   197	                ", downloadedPieces=" + downloadedPieces +
   198	                ", filesReceivedBytes=" + Arrays.toString(filesReceivedBytes) +
   199	                ", shareRatio=" + shareRatio +
   200	                ", activeTime=" + activeTime +
   201	                ", seedingTime=" + seedingTime +
   202	                ", availability=" + availability +
   203	                ", filesAvailability=" + Arrays.toString(filesAvailability) +
   204	                ", leechers=" + leechers +
   205	                ", totalLeechers=" + totalLeechers +
   206	                '}';
   207	    }
   208	}

    29	/*
    30	 * Provides full information about magnet.
    31	 */
    32	
    33	public class MagnetInfo implements Parcelable
    34	{
    35	    private String uri;
    36	    private String sha1hash;
    37	    private String name;
    38	    /*
    39	     * BEP53 standard http://www.bittorrent.org/beps/bep_0053.html
    40	     *
    41	     * Note: priorities number may not coincide with
    42	     *       the actual files number in the torrent.
    43	     *       In this case, manually add the missing number
    44	     *       of priorities to the end of the array
    45	     */
    46	    private List<Priority> filePriorities;
    47	
    48	    public MagnetInfo(String uri, String sha1hash, String name, List<Priority> filePriorities)
    49	    {
    50	        this.uri = uri;
    51	        this.sha1hash = sha1hash;
    52	        this.name = name;
    53	        this.filePriorities = filePriorities;
    54	    }
    55	
    56	    public MagnetInfo(Parcel s)
    57	    {
    58	        uri = s.readString();
    59	        sha1hash = s.readString();
    60	        name = s.readString();
    61	        filePriorities = s.readArrayList(Priority.class.getClassLoader());
    62	    }
    63	
    64	    @Override
    65	    public int describeContents()
    66	    {
    67	        return 0;
    68	    }
    69	
    70	    @Override
    71	    public void writeToParcel(Parcel dest, int flags)
    72	    {
    73	        dest.writeString(uri);
    74	        dest.writeString(sha1hash);
    75	        dest.writeString(name);
    76	        dest.writeList(filePriorities);
    77	    }
    78	
    79	    public static final Parcelable.Creator<MagnetInfo> CREATOR =
    80	            new Parcelable.Creator<MagnetInfo>()
    81	            {
    82	                @Override
    83	                public MagnetInfo createFromParcel(Parcel source)
    84	                {
    85	                    return new MagnetInfo(source);
    86	                }
    87	
    88	                @Override
    89	                public MagnetInfo[] newArray(int size)
    90	                {
    91	                    return new MagnetInfo[size];
    92	                }
    93	            };
    94	
    95	    public String getUri()
    96	    {
    97	        return uri;
    98	    }
    99	
   100	    public String getSha1hash()
   101	    {
   102	        return sha1hash;
   103	    }
   104	
   105	    public String getName()
   106	    {
   107	        return name;
   108	    }
   109	
   110	    public List<Priority> getFilePriorities()
   111	    {
   112	        return filePriorities;
   113	    }
   114	
   115	    @Override
   116	    public int hashCode()
   117	    {
   118	        return sha1hash.hashCode();
   119	    }
   120	
   121	    @NonNull
   122	    @Override
   123	    public String toString()
   124	    {
   125	        return "MagnetInfo{" +
   126	                "uri='" + uri + '\'' +
   127	                ", sha1hash='" + sha1hash + '\'' +
   128	                ", name='" + name + '\'' +
   129	                ", filePriorities=" + filePriorities +
   130	                '}';
   131	    }
   132	}

   116	        sha1Hash = source.readString();

   118	        createdBy = source.readString();
   119	        torrentSize = source.readLong();
   120	        creationDate = source.readLong();
   121	        fileCount = source.readInt();
   122	        fileList = new ArrayList<>();
   123	        source.readTypedList(fileList, BencodeFileItem.CREATOR);
