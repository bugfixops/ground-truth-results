   193	        if (url == null || url.isEmpty())

   196	        if (options == null)

   333	    private static Map<String, Collection<String>> parseQuery(String query)
   334	    {
   335	        HashMap<String, Collection<String>> queryPairs = new HashMap<>();
   336	
   337	        if (query == null || query.isEmpty() || query.equals("?"))
   338	            return queryPairs;
   339	
   340	        String[] pairs = query.split("&");
   341	        for (String pair : pairs) {
   342	            /* Ignore nested query string in the value of the pair, if one exists */
   343	            int equalPos = pair.indexOf('=');
   344	            if (equalPos <= 0)
   345	                continue;
   346	
   347	            String key = pair.substring(0, equalPos);
   348	            String value = pair.substring(equalPos + 1);
   349	
   350	            if (!key.isEmpty()) {
   351	                Collection<String> existing = queryPairs.get(key);
   352	                if (existing == null)
   353	                    existing = new ArrayList<>();
   354	
   355	                if (!value.isEmpty())
   356	                    existing.add(value);
   357	                queryPairs.put(key, existing);
   358	            }
   359	        }
   360	
   361	        return queryPairs;
   362	    }

   364	    private static String mapToQuery(Map<String, Collection<String>> query)
   365	    {
   366	        if (query.isEmpty())
   367	            return null;
   368	
   369	        StringBuilder sb = new StringBuilder();
   370	        int i = 0;
   371	        for (Map.Entry<String, Collection<String>> parameter : query.entrySet()) {
   372	            for (String value : parameter.getValue()) {
   373	                if (i > 0)
   374	                    sb.append("&");
   375	                sb.append(parameter.getKey());
   376	                if (value != null && !value.isEmpty()) {
   377	                    sb.append("=");
   378	                    sb.append(value);
   379	                }
   380	                i++;
   381	            }
   382	        }
   383	
   384	        return sb.toString();
   385	    }

   398	
   399	        if (host != null)

   401	        if (path != null && !path.isEmpty())

   408	            output.append('?').append(query);
   409	
   410	        if (hash != null && !hash.isEmpty())
   411	            output.append('#').append(hash);
   412	
   413	        return output.toString();

    47	        String merged;
    48	
    49	        if (ref == null || ref.isEmpty()) {
    50	            merged = base;
    51	        } else if (ref.charAt(0) != '/' && base != null && !base.isEmpty()) {
    52	            int i = base.lastIndexOf("/");
    53	            merged = base.substring(0, i + 1) + ref;
    54	        } else {
    55	            merged = ref;
    56	        }
    57

    60	        }

    74	            return "";
    75	        }

    77	        List<String> result = new ArrayList<>();
    78	
    79	        for (String part : parts) {
    80	            switch (part) {
    81	                case "":
    82	                case ".":
    83	                    // Ignore
    84	                    break;
    85	                case "..":
    86	                    if (result.size() > 0) {
    87	                        result.remove(result.size() - 1);
    88	                    }
    89	                    break;
    90	                default:
    91	                    result.add(part);
    92	                    break;
    93	            }
    94	        }
    95	
    96	        // Get last element, if it was '.' or '..' we need
    97	        // to end in a slash.
    98	        switch (parts[parts.length - 1]) {
    99	            case ".":
   100	            case "..":
   101	                // Add an empty last string, it will be turned into
   102	                // a slash when joined together.
   103	                result.add("");
   104	                break;
   105	        }
   106	
   107	        return "/" + join("/", result);
   108	    }

   110	    private static String join(CharSequence delimiter, List<String> tokens)
   111	    {
   112	        int length = tokens.size();
   113	        if (length == 0)
   114	            return "";
   115	
   116	        StringBuilder sb = new StringBuilder();
   117	        sb.append(tokens.get(0));
   118	        for (int i = 1; i < length; i++) {
   119	            sb.append(delimiter);
   120	            sb.append(tokens.get(i));
   121	        }
   122	
   123	        return sb.toString();
   124	    }
