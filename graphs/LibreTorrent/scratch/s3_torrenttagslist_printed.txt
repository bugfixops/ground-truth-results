     1	package org.proninyaroslav.libretorrent.ui.tag;
     2	
     3	import android.content.Context;
     4	import android.content.res.ColorStateList;
     5	import android.util.AttributeSet;
     6	import android.util.TypedValue;
     7	
     8	import androidx.annotation.NonNull;
     9	
    10	import com.google.android.material.chip.Chip;
    11	import com.google.android.material.chip.ChipGroup;
    12	
    13	import org.proninyaroslav.libretorrent.R;
    14	import org.proninyaroslav.libretorrent.core.model.data.entity.TagInfo;
    15	
    16	import java.util.List;
    17	
    18	public class TorrentTagsList extends ChipGroup {
    19	    private Listener listener;
    20	    private List<TagInfo> tags;
    21	    private AddTagButton addTagButton;
    22	
    23	    public TorrentTagsList(Context context) {
    24	        super(context);
    25	
    26	        init(context);
    27	    }
    28	
    29	    public TorrentTagsList(Context context, AttributeSet attrs) {
    30	        super(context, attrs);
    31	
    32	        init(context);
    33	    }
    34	
    35	    void init(Context context) {
    36	        this.addTagButton = buildAddTagButton(context);
    37	        addView(addTagButton);
    38	    }
    39	
    40	    public void setListener(Listener listener) {
    41	        this.listener = listener;
    42	    }
    43	
    44	    public void submit(@NonNull List<TagInfo> tags) {
    45	        removeAllViews();
    46	        this.tags = tags;
    47	
    48	        for (TagInfo tag : tags) {
    49	            addView(buildChip(getContext(), tag));
    50	        }
    51	        addView(addTagButton);
    52	    }
    53	
    54	    private Chip buildChip(Context context, TagInfo info) {
    55	        Chip chip = new Chip(context);
    56	        chip.setText(info.name);
    57	        chip.setChipIconSize(TypedValue.applyDimension(
    58	                TypedValue.COMPLEX_UNIT_DIP,
    59	                18,
    60	                getResources().getDisplayMetrics()
    61	        ));
    62	        chip.setChipIconResource(R.drawable.torrent_tag_color_background);
    63	        chip.setChipIconTint(new ColorStateList(
    64	                new int[][]{new int[]{}},
    65	                new int[]{info.color})
    66	        );
    67	        chip.setCloseIconVisible(true);
    68	        chip.setClickable(false);
    69	        chip.setOnCloseIconClickListener((v) -> {
    70	            if (listener != null) {
    71	                listener.onTagRemoved(info);
    72	            }
    73	        });
    74	
    75	        return chip;
    76	    }
    77	
    78	    private AddTagButton buildAddTagButton(Context context) {
    79	        AddTagButton button = new AddTagButton(context);
    80	        button.setOnClickListener((v) -> {
    81	            if (listener != null) {
    82	                listener.onAddTagClick();
    83	            }
    84	        });
    85	
    86	        return button;
    87	    }
    88	
    89	    public interface Listener {
    90	        void onAddTagClick();
    91	
    92	        void onTagRemoved(@NonNull TagInfo info);
    93	    }
    94	}
