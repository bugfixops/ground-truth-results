    20	package org.proninyaroslav.libretorrent.ui.settings;
    21	
    22	import android.content.Context;
    23	import android.content.Intent;
    24	import android.os.Bundle;
    25	import android.view.View;
    26	
    27	import androidx.annotation.NonNull;
    28	import androidx.annotation.Nullable;
    29	import androidx.appcompat.app.AppCompatActivity;
    30	import androidx.fragment.app.Fragment;
    31	import androidx.fragment.app.FragmentTransaction;
    32	import androidx.lifecycle.ViewModelProvider;
    33	import androidx.preference.Preference;
    34	import androidx.preference.PreferenceFragmentCompat;
    35	
    36	import org.proninyaroslav.libretorrent.R;
    37	import org.proninyaroslav.libretorrent.core.utils.Utils;
    38	import org.proninyaroslav.libretorrent.ui.settings.sections.AppearanceSettingsFragment;
    39	import org.proninyaroslav.libretorrent.ui.settings.sections.BehaviorSettingsFragment;
    40	import org.proninyaroslav.libretorrent.ui.settings.sections.FeedSettingsFragment;
    41	import org.proninyaroslav.libretorrent.ui.settings.sections.LimitationsSettingsFragment;
    42	import org.proninyaroslav.libretorrent.ui.settings.sections.NetworkSettingsFragment;
    43	import org.proninyaroslav.libretorrent.ui.settings.sections.SchedulingSettingsFragment;
    44	import org.proninyaroslav.libretorrent.ui.settings.sections.StorageSettingsFragment;
    45	import org.proninyaroslav.libretorrent.ui.settings.sections.StreamingSettingsFragment;
    46	
    47	public class SettingsFragment extends PreferenceFragmentCompat
    48	{
    49	    private static final String TAG = SettingsFragment.class.getSimpleName();
    50	
    51	    private static final String AppearanceSettings = "AppearanceSettingsFragment";
    52	    private static final String BehaviorSettings = "BehaviorSettingsFragment";
    53	    private static final String NetworkSettings = "NetworkSettingsFragment";
    54	    private static final String StorageSettings = "StorageSettingsFragment";
    55	    private static final String LimitationsSettings = "LimitationsSettingsFragment";
    56	    private static final String SchedulingSettings = "SchedulingSettingsFragment";
    57	    private static final String FeedSettings = "FeedSettingsFragment";
    58	    private static final String StreamingSettings = "StreamingSettingsFragment";
    59	
    60	    private AppCompatActivity activity;
    61	    private SettingsViewModel viewModel;
    62	
    63	    public static SettingsFragment newInstance()
    64	    {
    65	        SettingsFragment fragment = new SettingsFragment();
    66	
    67	        fragment.setArguments(new Bundle());
    68	
    69	        return fragment;
    70	    }
    71	
    72	    @Override
    73	    public void onAttach(@NonNull Context context)
    74	    {
    75	        super.onAttach(context);
    76	
    77	        if (context instanceof AppCompatActivity)
    78	            activity = (AppCompatActivity)context;
    79	    }
    80	
    81	    @Override
    82	    public void onViewCreated(@NonNull View view, @Nullable Bundle savedInstanceState) {
    83	        super.onViewCreated(view, savedInstanceState);
    84	
    85	        if (activity == null)
    86	            activity = (AppCompatActivity)getActivity();
    87	
    88	        viewModel = new ViewModelProvider(activity).get(SettingsViewModel.class);
    89	
    90	        if (Utils.isTwoPane(activity)) {
    91	            Fragment f = activity.getSupportFragmentManager()
    92	                    .findFragmentById(R.id.detail_fragment_container);
    93	            if (f == null)
    94	                setFragment(AppearanceSettingsFragment.newInstance(),
    95	                        getString(R.string.pref_header_appearance));
    96	        }
    97	
    98	        Preference appearance = findPreference(AppearanceSettingsFragment.class.getSimpleName());
    99	        if (appearance != null)
   100	            appearance.setOnPreferenceClickListener(prefClickListener);
   101	
   102	        Preference behavior = findPreference(BehaviorSettingsFragment.class.getSimpleName());
   103	        if (behavior != null)
   104	            behavior.setOnPreferenceClickListener(prefClickListener);
   105	
   106	        Preference storage = findPreference(StorageSettingsFragment.class.getSimpleName());
   107	        if (storage != null)
   108	            storage.setOnPreferenceClickListener(prefClickListener);
   109	
   110	        Preference limitations = findPreference(LimitationsSettingsFragment.class.getSimpleName());
   111	        if (limitations != null)
   112	            limitations.setOnPreferenceClickListener(prefClickListener);
   113	
   114	        Preference network = findPreference(NetworkSettingsFragment.class.getSimpleName());
   115	        if (network != null)
   116	            network.setOnPreferenceClickListener(prefClickListener);
   117	
   118	        Preference scheduling = findPreference(SchedulingSettingsFragment.class.getSimpleName());
   119	        if (scheduling != null)
   120	            scheduling.setOnPreferenceClickListener(prefClickListener);
   121	
   122	        Preference feed = findPreference(FeedSettingsFragment.class.getSimpleName());
   123	        if (feed != null)
   124	            feed.setOnPreferenceClickListener(prefClickListener);
   125	
   126	        Preference streaming = findPreference(StreamingSettingsFragment.class.getSimpleName());
   127	        if (streaming != null)
   128	            streaming.setOnPreferenceClickListener(prefClickListener);
   129	    }
   130	
   131	    private Preference.OnPreferenceClickListener prefClickListener = (preference) -> {
   132	        openPreference(preference.getKey());
   133	        return true;
   134	    };
   135	
   136	    private void openPreference(String prefName)
   137	    {
   138	        switch (prefName) {
   139	            case AppearanceSettings:
   140	                if (Utils.isLargeScreenDevice(activity)) {
   141	                    setFragment(AppearanceSettingsFragment.newInstance(),
   142	                            getString(R.string.pref_header_appearance));
   143	                } else {
   144	                    startActivity(AppearanceSettingsFragment.class,
   145	                            getString(R.string.pref_header_appearance));
   146	                }
   147	                break;
   148	            case BehaviorSettings:
   149	                if (Utils.isLargeScreenDevice(activity)) {
   150	                    setFragment(BehaviorSettingsFragment.newInstance(),
   151	                            getString(R.string.pref_header_behavior));
   152	                } else {
   153	                    startActivity(BehaviorSettingsFragment.class,
   154	                            getString(R.string.pref_header_behavior));
   155	                }
   156	                break;
   157	            case NetworkSettings:
   158	                if (Utils.isLargeScreenDevice(activity)) {
   159	                    setFragment(NetworkSettingsFragment.newInstance(),
   160	                            getString(R.string.pref_header_network));
   161	                } else {
   162	                    startActivity(NetworkSettingsFragment.class,
   163	                            getString(R.string.pref_header_network));
   164	                }
   165	                break;
   166	            case StorageSettings:
   167	                if (Utils.isLargeScreenDevice(activity)) {
   168	                    setFragment(StorageSettingsFragment.newInstance(),
   169	                            getString(R.string.pref_header_storage));
   170	                } else {
   171	                    startActivity(StorageSettingsFragment.class,
   172	                            getString(R.string.pref_header_storage));
   173	                }
   174	                break;
   175	            case LimitationsSettings:
   176	                if (Utils.isLargeScreenDevice(activity)) {
   177	                    setFragment(LimitationsSettingsFragment.newInstance(),
   178	                            getString(R.string.pref_header_limitations));
   179	                } else {
   180	                    startActivity(LimitationsSettingsFragment.class,
   181	                            getString(R.string.pref_header_limitations));
   182	                }
   183	                break;
   184	            case SchedulingSettings:
   185	                if (Utils.isLargeScreenDevice(activity)) {
   186	                    setFragment(SchedulingSettingsFragment.newInstance(),
   187	                            getString(R.string.pref_header_scheduling));
   188	                } else {
   189	                    startActivity(SchedulingSettingsFragment.class,
   190	                            getString(R.string.pref_header_scheduling));
   191	                }
   192	                break;
   193	            case FeedSettings:
   194	                if (Utils.isLargeScreenDevice(activity)) {
   195	                    setFragment(FeedSettingsFragment.newInstance(),
   196	                            getString(R.string.pref_header_feed));
   197	                } else {
   198	                    startActivity(FeedSettingsFragment.class,
   199	                            getString(R.string.pref_header_feed));
   200	                }
   201	                break;
   202	            case StreamingSettings:
   203	                if (Utils.isLargeScreenDevice(activity)) {
   204	                    setFragment(StreamingSettingsFragment.newInstance(),
   205	                            getString(R.string.pref_header_streaming));
   206	                } else {
   207	                    startActivity(StreamingSettingsFragment.class,
   208	                            getString(R.string.pref_header_streaming));
   209	                }
   210	                break;
   211	        }
   212	    }
   213	
   214	    @Override
   215	    public void onCreatePreferences(Bundle savedInstanceState, String rootKey)
   216	    {
   217	        setPreferencesFromResource(R.xml.pref_headers, rootKey);
   218	    }
   219	
   220	    private <F extends PreferenceFragmentCompat> void setFragment(F fragment, String title)
   221	    {
   222	        viewModel.detailTitleChanged.setValue(title);
   223	
   224	        if (Utils.isLargeScreenDevice(activity)) {
   225	            activity.getSupportFragmentManager().beginTransaction()
   226	                    .replace(R.id.detail_fragment_container, fragment)
   227	                    .setTransition(FragmentTransaction.TRANSIT_FRAGMENT_FADE)
   228	                    .commit();
   229	        }
   230	    }
   231	
   232	    private <F extends PreferenceFragmentCompat> void startActivity(Class<F> fragment, String title)
   233	    {
   234	        Intent i = new Intent(activity, PreferenceActivity.class);
   235	        PreferenceActivityConfig config = new PreferenceActivityConfig(
   236	                fragment.getSimpleName(),
   237	                title);
   238	
   239	        i.putExtra(PreferenceActivity.TAG_CONFIG, config);
   240	        startActivity(i);
   241	    }
   242	}
