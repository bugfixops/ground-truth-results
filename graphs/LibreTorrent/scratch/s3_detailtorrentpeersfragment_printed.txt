    20	package org.proninyaroslav.libretorrent.ui.detailtorrent.pages.peers;
    21	
    22	import android.content.Context;
    23	import android.content.Intent;
    24	import android.content.res.TypedArray;
    25	import android.os.Bundle;
    26	import android.os.Parcelable;
    27	import android.util.TypedValue;
    28	import android.view.LayoutInflater;
    29	import android.view.View;
    30	import android.view.ViewGroup;
    31	
    32	import androidx.annotation.NonNull;
    33	import androidx.annotation.Nullable;
    34	import androidx.appcompat.app.AppCompatActivity;
    35	import androidx.databinding.DataBindingUtil;
    36	import androidx.fragment.app.Fragment;
    37	import androidx.lifecycle.ViewModelProvider;
    38	import androidx.recyclerview.widget.DefaultItemAnimator;
    39	import androidx.recyclerview.widget.LinearLayoutManager;
    40	import androidx.recyclerview.widget.RecyclerView;
    41	
    42	import org.proninyaroslav.libretorrent.R;
    43	import org.proninyaroslav.libretorrent.databinding.FragmentDetailTorrentPeerListBinding;
    44	import org.proninyaroslav.libretorrent.ui.customviews.RecyclerViewDividerDecoration;
    45	import org.proninyaroslav.libretorrent.ui.detailtorrent.DetailTorrentViewModel;
    46	
    47	import io.reactivex.Flowable;
    48	import io.reactivex.android.schedulers.AndroidSchedulers;
    49	import io.reactivex.disposables.CompositeDisposable;
    50	import io.reactivex.schedulers.Schedulers;
    51	
    52	/*
    53	 * The fragment for displaying bittorrent peer list. Part of DetailTorrentFragment.
    54	 */
    55	
    56	public class DetailTorrentPeersFragment extends Fragment
    57	    implements PeerListAdapter.ClickListener
    58	{
    59	    private static final String TAG = DetailTorrentPeersFragment.class.getSimpleName();
    60	
    61	    private static final String TAG_LIST_TRACKER_STATE = "list_tracker_state";
    62	
    63	    private AppCompatActivity activity;
    64	    private FragmentDetailTorrentPeerListBinding binding;
    65	    private DetailTorrentViewModel viewModel;
    66	    private LinearLayoutManager layoutManager;
    67	    private PeerListAdapter adapter;
    68	    /* Save state scrolling */
    69	    private Parcelable listPeerState;
    70	    private CompositeDisposable disposables = new CompositeDisposable();
    71	
    72	    public static DetailTorrentPeersFragment newInstance()
    73	    {
    74	        DetailTorrentPeersFragment fragment = new DetailTorrentPeersFragment();
    75	
    76	        fragment.setArguments(new Bundle());
    77	
    78	        return fragment;
    79	    }
    80	
    81	    @Override
    82	    public View onCreateView(@NonNull LayoutInflater inflater, ViewGroup container, Bundle savedInstanceState)
    83	    {
    84	        binding = DataBindingUtil.inflate(inflater, R.layout.fragment_detail_torrent_peer_list, container, false);
    85	
    86	        return binding.getRoot();
    87	    }
    88	
    89	    @Override
    90	    public void onAttach(@NonNull Context context)
    91	    {
    92	        super.onAttach(context);
    93	
    94	        if (context instanceof AppCompatActivity)
    95	            activity = (AppCompatActivity)context;
    96	    }
    97	
    98	    @Override
    99	    public void onStop()
   100	    {
   101	        super.onStop();
   102	
   103	        disposables.clear();
   104	    }
   105	
   106	    @Override
   107	    public void onStart()
   108	    {
   109	        super.onStart();
   110	
   111	        subscribeAdapter();
   112	    }
   113	
   114	    @Override
   115	    public void onViewCreated(@NonNull View view, @Nullable Bundle savedInstanceState) {
   116	        super.onViewCreated(view, savedInstanceState);
   117	
   118	        if (activity == null)
   119	            activity = (AppCompatActivity) getActivity();
   120	
   121	        viewModel = new ViewModelProvider(activity).get(DetailTorrentViewModel.class);
   122	
   123	        layoutManager = new LinearLayoutManager(activity);
   124	        binding.peerList.setLayoutManager(layoutManager);
   125	        binding.peerList.setEmptyView(binding.emptyViewPeerList);
   126	        adapter = new PeerListAdapter(this);
   127	        /*
   128	         * A RecyclerView by default creates another copy of the ViewHolder in order to
   129	         * fade the views into each other. This causes the problem because the old ViewHolder gets
   130	         * the payload but then the new one doesn't. So needs to explicitly tell it to reuse the old one.
   131	         */
   132	        DefaultItemAnimator animator = new DefaultItemAnimator() {
   133	            @Override
   134	            public boolean canReuseUpdatedViewHolder(@NonNull RecyclerView.ViewHolder viewHolder) {
   135	                return true;
   136	            }
   137	        };
   138	        binding.peerList.setItemAnimator(animator);
   139	        TypedArray a = activity.obtainStyledAttributes(new TypedValue().data, new int[]{R.attr.divider});
   140	        binding.peerList.addItemDecoration(new RecyclerViewDividerDecoration(a.getDrawable(0)));
   141	        a.recycle();
   142	        binding.peerList.setAdapter(adapter);
   143	    }
   144	
   145	    @Override
   146	    public void onSaveInstanceState(@NonNull Bundle outState)
   147	    {
   148	        listPeerState = layoutManager.onSaveInstanceState();
   149	        outState.putParcelable(TAG_LIST_TRACKER_STATE, listPeerState);
   150	
   151	        super.onSaveInstanceState(outState);
   152	    }
   153	
   154	    @Override
   155	    public void onViewStateRestored(Bundle savedInstanceState)
   156	    {
   157	        super.onViewStateRestored(savedInstanceState);
   158	
   159	        if (savedInstanceState != null)
   160	            listPeerState = savedInstanceState.getParcelable(TAG_LIST_TRACKER_STATE);
   161	    }
   162	
   163	    @Override
   164	    public void onResume()
   165	    {
   166	        super.onResume();
   167	
   168	        if (listPeerState != null)
   169	            layoutManager.onRestoreInstanceState(listPeerState);
   170	    }
   171	
   172	    private void subscribeAdapter()
   173	    {
   174	        disposables.add(viewModel.observePeers()
   175	                .subscribeOn(Schedulers.io())
   176	                .flatMapSingle((children) ->
   177	                        Flowable.fromIterable(children)
   178	                                .map(PeerItem::new)
   179	                                .toList()
   180	                )
   181	                .observeOn(AndroidSchedulers.mainThread())
   182	                .subscribe((children) -> adapter.submitList(children)));
   183	    }
   184	
   185	    @Override
   186	    public boolean onItemLongClick(@NonNull PeerItem item)
   187	    {
   188	        sharePeerIp(item.ip);
   189	
   190	        return true;
   191	    }
   192	
   193	    private void sharePeerIp(String ip)
   194	    {
   195	        Intent sharingIntent = new Intent(android.content.Intent.ACTION_SEND);
   196	        sharingIntent.setType("text/plain");
   197	        sharingIntent.putExtra(android.content.Intent.EXTRA_SUBJECT, "ip");
   198	        sharingIntent.putExtra(android.content.Intent.EXTRA_TEXT, ip);
   199	
   200	        startActivity(Intent.createChooser(sharingIntent, getString(R.string.share_via)));
   201	    }
   202	}
