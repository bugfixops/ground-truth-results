   653	    private void showDetailTorrent(String id) {
   654	        if (Utils.isTwoPane(this)) {
   655	            FragmentManager fm = getSupportFragmentManager();
   656	            DetailTorrentFragment detail = DetailTorrentFragment.newInstance(id);
   657	            Fragment fragment = fm.findFragmentById(R.id.detail_torrent_fragmentContainer);
   658	
   659	            if (fragment instanceof DetailTorrentFragment) {
   660	                String oldId = ((DetailTorrentFragment) fragment).getTorrentId();
   661	                if (id.equals(oldId))
   662	                    return;
   663	            }
   664	            fm.beginTransaction()
   665	                    .replace(R.id.detail_torrent_fragmentContainer, detail)
   666	                    .setTransition(FragmentTransaction.TRANSIT_FRAGMENT_FADE)
   667	                    .commit();
   668	
   669	        } else {
   670	            Intent i = new Intent(this, DetailTorrentActivity.class);
   671	            i.putExtra(DetailTorrentActivity.TAG_TORRENT_ID, id);
   672	            startActivity(i);
   673	        }
