    20	package org.proninyaroslav.libretorrent.ui.feeditems;
    21	
    22	import android.content.Intent;
    23	import android.os.Bundle;
    24	
    25	import androidx.annotation.NonNull;
    26	import androidx.annotation.Nullable;
    27	import androidx.appcompat.app.AppCompatActivity;
    28	import androidx.fragment.app.Fragment;
    29	
    30	import org.proninyaroslav.libretorrent.R;
    31	import org.proninyaroslav.libretorrent.core.utils.Utils;
    32	import org.proninyaroslav.libretorrent.ui.FragmentCallback;
    33	
    34	public class FeedItemsActivity extends AppCompatActivity
    35	        implements FragmentCallback
    36	{
    37	    private static final String TAG = FeedItemsActivity.class.getSimpleName();
    38	
    39	    public static final String TAG_FEED_ID = "feed_id";
    40	
    41	    @Override
    42	    protected void onCreate(@Nullable Bundle savedInstanceState)
    43	    {
    44	        setTheme(Utils.getAppTheme(getApplicationContext()));
    45	        super.onCreate(savedInstanceState);
    46	
    47	        if (Utils.isTwoPane(this)) {
    48	            finish();
    49	            return;
    50	        }
    51	
    52	        setContentView(R.layout.activity_feed_items);
    53	
    54	        FeedItemsFragment feedItemsFragment = (FeedItemsFragment)getSupportFragmentManager()
    55	                .findFragmentById(R.id.feed_items_fragmentContainer);
    56	
    57	        if (feedItemsFragment != null)
    58	            feedItemsFragment.setFeedId(getIntent().getLongExtra(TAG_FEED_ID, -1));
    59	    }
    60	
    61	    @Override
    62	    public void onFragmentFinished(@NonNull Fragment f, Intent intent,
    63	                                   @NonNull FragmentCallback.ResultCode code)
    64	    {
    65	        finish();
    66	    }
    67	}
