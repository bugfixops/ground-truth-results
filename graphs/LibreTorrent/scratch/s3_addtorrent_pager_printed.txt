   136	
   137	    final ActivityResultLauncher<Intent> selectTag = registerForActivityResult(
   138	            new ActivityResultContracts.StartActivityForResult(),
   139	            (result) -> {
   140	                Intent data = result.getData();
   141	                if (result.getResultCode() == Activity.RESULT_OK && data != null) {
   142	                    TagInfo tag = data.getParcelableExtra(SelectTagActivity.TAG_RESULT_SELECTED_TAG);
   143	                    if (tag != null) {
   144	                        viewModel.addTag(tag);
   145	                    }
   146	                }
   147	            }
   148	    );
   149	
   150	    final ActivityResultLauncher<Intent> chooseDir = registerForActivityResult(
   151	            new ActivityResultContracts.StartActivityForResult(), (result) -> {
   152	                if (result.getResultCode() != Activity.RESULT_OK) {
   153	                    return;
   154	                }
   155	                Intent data = result.getData();
   156	                if (data == null || data.getData() == null) {
   157	                    showOpenDirErrorDialog();
   158	                    return;
   159	                }
   160	                viewModel.mutableParams.getDirPath().set(data.getData());
   161	            });
   162	
   163	    private final TorrentTagsList.Listener tagsListener = new TorrentTagsList.Listener() {
   164	        @Override
   165	        public void onAddTagClick() {
   166	            disposables.add(Observable.fromIterable(viewModel.getCurrentTags())
   167	                    .map((tag) -> tag.id)
   168	                    .toList()
   169	                    .subscribe((ids) -> {
   170	                        Intent i = new Intent(activity, SelectTagActivity.class);
   171	                        i.putExtra(
   172	                                SelectTagActivity.TAG_EXCLUDE_TAGS_ID,
   173	                                ArrayUtils.toPrimitive(ids.toArray(new Long[0]))
   174	                        );
   175	                        selectTag.launch(i);
   176	                    })
   177	            );
   178	        }
   179	
   180	        @Override
   181	        public void onTagRemoved(@NonNull TagInfo info) {
   182	            viewModel.removeTag(info);
   183	        }
   184	    };
   185	
   186	    private void subscribeTags() {
   187	        disposables.add(viewModel.observeTags()
   188	                .subscribe(binding.info.tagsList::submit)
   189	        );
   190	    }
   191	
   192	    private void showChooseDirDialog() {
   193	        Intent i = new Intent(activity, FileManagerDialog.class);
   194	
   195	        FileSystemFacade fs = SystemFacadeHelper.getFileSystemFacade(
   196	                activity.getApplicationContext()
   197	        );
   198	        String path;
   199	        try {
   200	            path = fs.getDirPath(viewModel.mutableParams.getDirPath().get());
   201	        } catch (UnknownUriException e) {
   202	            path = null;
   203	        }
   204	
   205	        FileManagerConfig config = new FileManagerConfig(
   206	                path,
   207	                getString(R.string.select_folder_to_save),
   208	                FileManagerConfig.DIR_CHOOSER_MODE
   209	        );
   210	
   211	        i.putExtra(FileManagerDialog.TAG_CONFIG, config);
   212	        chooseDir.launch(i);
   213	    }
   214	
   215	    private void showOpenDirErrorDialog() {
   216	        if (!isAdded())
   217	            return;
   218	
   219	        FragmentManager fm = getChildFragmentManager();
   220	        if (fm.findFragmentByTag(TAG_OPEN_DIR_ERROR_DIALOG) == null) {
   221	            BaseAlertDialog openDirErrorDialog = BaseAlertDialog.newInstance(
   222	                    getString(R.string.error),
   223	                    getString(R.string.unable_to_open_folder),
   224	                    0,
   225	                    getString(R.string.ok),
   226	                    null,
   227	                    null,
   228	                    true);
   229	
   230	            openDirErrorDialog.show(fm, TAG_OPEN_DIR_ERROR_DIALOG);
   231	        }
   232	    }

   234	    private void checkNameField(Editable s) {
   235	        if (TextUtils.isEmpty(s)) {
   236	            binding.info.layoutName.setErrorEnabled(true);
   237	            binding.info.layoutName.setError(getString(R.string.error_field_required));
   238	            binding.info.layoutName.requestFocus();
   239	        } else {
   240	            binding.info.layoutName.setErrorEnabled(false);
   241	            binding.info.layoutName.setError(null);
   242	        }
   243	    }

   111	    private void subscribeAdapter()
   112	    {
   113	        disposable.add(viewModel.children
   114	                .subscribeOn(Schedulers.computation())
   115	                .flatMapSingle((children) ->
   116	                        Flowable.fromIterable(children)
   117	                                .map(DownloadableFileItem::new)
   118	                                .toList()
   119	                )
   120	                .observeOn(AndroidSchedulers.mainThread())
   121	                .subscribe((children) -> {
   122	                    adapter.submitList(children);
   123	                    updateFileSize();
   124	                }));
   125	    }

    74	    public static final DiffUtil.ItemCallback<DownloadableFileItem> diffCallback = new DiffUtil.ItemCallback<DownloadableFileItem>()
    75	    {
    76	        @Override
    77	        public boolean areContentsTheSame(@NonNull DownloadableFileItem oldItem,

    84	        public boolean areItemsTheSame(@NonNull DownloadableFileItem oldItem,

    88	        }

    91	    public static class ViewHolder extends RecyclerView.ViewHolder
    92	    {
    93	        private ItemTorrentDownloadableFileBinding binding;
    94	
    95	        public ViewHolder(ItemTorrentDownloadableFileBinding binding)
    96	        {
    97	            super(binding.getRoot());
    98	
    99	            this.binding = binding;
   100	        }
   101	
   102	        void bind(DownloadableFileItem item, ClickListener listener)
   103	        {
   104	            Context context = itemView.getContext();
   105	
   106	            itemView.setOnClickListener((v) -> {
   107	                if (item.isFile) {
   108	                    /* Check file if it clicked */
   109	                    binding.selected.performClick();
   110	                }
   111	                if (listener != null)
   112	                    listener.onItemClicked(item);
   113	            });
   114	            binding.selected.setOnClickListener((View v) -> {
   115	                if (listener != null)
   116	                    listener.onItemCheckedChanged(item, binding.selected.isChecked());
   117	            });
   118	
   119	            binding.name.setText(item.name);
   120	
   121	            boolean isParentDir = item.name.equals(BencodeFileTree.PARENT_DIR);
   122	
   123	            if (item.isFile) {
   124	                binding.icon.setImageResource(R.drawable.ic_file_grey600_24dp);
   125	                binding.icon.setContentDescription(context.getString(R.string.file));
   126	
   127	            } else {
   128	                if (isParentDir) {
   129	                    binding.icon.setImageResource(R.drawable.ic_arrow_up_bold_grey600_24dp);
   130	                    binding.icon.setContentDescription(context.getString(R.string.parent_folder));
   131	                } else {
   132	                    binding.icon.setImageResource(R.drawable.ic_folder_grey600_24dp);
   133	                    binding.icon.setContentDescription(context.getString(R.string.folder));
   134	                }

   202	    private static class TorrentDecodeTask extends AsyncTask<Uri, Void, Throwable> {
   203	        private final WeakReference<AddTorrentViewModel> viewModel;
   204	
   205	        private TorrentDecodeTask(AddTorrentViewModel viewModel) {
   206	            super();
   207	
   208	            this.viewModel = new WeakReference<>(viewModel);
   209	        }
   210	
   211	        @Override
   212	        protected void onPreExecute() {
   213	            /* Nothing */
   214	        }
   215	
   216	        @Override
   217	        protected Throwable doInBackground(Uri... params) {
   218	            AddTorrentViewModel v = viewModel.get();
   219	            if (v == null || isCancelled())
   220	                return null;
   221	
   222	            Uri uri = params[0];
   223	            try {
   224	                switch (uri.getScheme()) {
   225	                    case Utils.FILE_PREFIX:
   226	                    case Utils.CONTENT_PREFIX:
   227	                        v.mutableParams.setSource(uri.toString());
   228	                        v.decodeState.postValue(
   229	                                new DecodeState(AddTorrentViewModel.Status.DECODE_TORRENT_FILE));
   230	                        break;
   231	                    case Utils.MAGNET_PREFIX:
   232	                        v.mutableParams.setSource(uri.toString());
   233	                        v.decodeState.postValue(
   234	                                new DecodeState(AddTorrentViewModel.Status.FETCHING_MAGNET));
   235	                        v.mutableParams.setFromMagnet(true);
   236	
   237	                        Pair<MagnetInfo, Single<TorrentMetaInfo>> res = v.engine.fetchMagnet(uri.toString());
   238	                        if (res == null)
   239	                            break;
   240	                        MagnetInfo magnetInfo = res.first;
   241	                        if (magnetInfo != null && !isCancelled()) {
   242	                            v.info.set(new TorrentMetaInfo(magnetInfo.getName(), magnetInfo.getSha1hash()));
   243	                            v.observeFetchedMetadata(res.second);
   244	
   245	                            if (magnetInfo.getFilePriorities() != null)
   246	                                v.magnetPriorities = new ArrayList<>(magnetInfo.getFilePriorities());
   247	                        }
   248	                        break;
   249	                    case Utils.HTTP_PREFIX:
   250	                    case Utils.HTTPS_PREFIX:
   251	                        v.decodeState.postValue(new DecodeState(AddTorrentViewModel.Status.FETCHING_HTTP));
   252	
   253	                        File httpTmp = v.fs.makeTempFile(".torrent");
   254	                        byte[] response = Utils.fetchHttpUrl(v.getApplication(), uri.toString());
   255	                        org.apache.commons.io.FileUtils.writeByteArrayToFile(httpTmp, response);
   256	
   257	                        if (httpTmp.exists() && !isCancelled()) {
   258	                            v.mutableParams.setSource(
   259	                                    v.fs.normalizeFileSystemPath(httpTmp.getAbsolutePath()));
   260	                        } else {
   261	                            return new IllegalArgumentException("Unknown path to the torrent file");
   262	                        }
   263	
   264	                        break;
   265	                    default:
   266	                        throw new IllegalArgumentException("Invalid scheme");
   267	                }
   268	
   269	                String tmpSource = v.mutableParams.getSource();
   270	                boolean fromMagnet = v.mutableParams.isFromMagnet();
   271	                if (tmpSource != null && !fromMagnet && !isCancelled())
   272	                    readTorrentFile(Uri.parse(tmpSource));
   273	
   274	            } catch (Throwable e) {
   275	                return e;
   276	            }
   277	
   278	            return null;
   279	        }
   280	
   281	        private void readTorrentFile(Uri uri) throws IOException, DecodeException {
   282	            AddTorrentViewModel v = viewModel.get();
   283	            if (v == null || isCancelled())
   284	                return;
   285	
   286	            ContentResolver contentResolver = v.getApplication().getContentResolver();
   287	            try (ParcelFileDescriptor outPfd = contentResolver.openFileDescriptor(uri, "r")) {
   288	                if (outPfd == null) {
   289	                    throw new IOException("ParcelFileDescriptor is null");
   290	                }
   291	                FileDescriptor outFd = outPfd.getFileDescriptor();
   292	                try (FileInputStream is = new FileInputStream(outFd)) {
   293	                    v.info.set(new TorrentMetaInfo(is));
   294	                }
   295	            } catch (FileNotFoundException e) {
   296	                throw new FileNotFoundException(uri.toString() + ": " + e.getMessage());
   297	            }
   298	        }
   299	
   300	        @Override
   301	        protected void onPostExecute(Throwable e) {
   302	            AddTorrentViewModel v = viewModel.get();
   303	            if (v == null || isCancelled())
   304	                return;
   305	
   306	            if (e != null) {
   307	                v.decodeState.postValue(new DecodeState(AddTorrentViewModel.Status.ERROR, e));
   308	                return;
   309	            }
   310	
   311	            DecodeState prevState = v.decodeState.getValue();
   312	            if (prevState == null)
   313	                return;
   314	
   315	            switch (prevState.status) {
   316	                case DECODE_TORRENT_FILE:
   317	                    v.decodeState.postValue(
   318	                            new DecodeState(AddTorrentViewModel.Status.DECODE_TORRENT_COMPLETED));
   319	                    break;
   320	                case FETCHING_HTTP:
   321	                    v.decodeState.postValue(
   322	                            new DecodeState(AddTorrentViewModel.Status.FETCHING_HTTP_COMPLETED));
   323	                    break;
   324	            }
   325	        }
   326	    }

   368	    public void makeFileTree() {
   369	        if (fileTree != null)
   370	            return;
   371	
   372	        TorrentMetaInfo infoObj = info.get();
   373	        if (infoObj == null)
   374	            return;
   375	        List<BencodeFileItem> files = infoObj.fileList;
   376	        if (files.isEmpty())
   377	            return;
   378	
   379	        Pair<BencodeFileTree, BencodeFileTree[]> res = BencodeFileTreeUtils.buildFileTree(files);
   380	        this.fileTree = res.first;
   381	        this.treeLeaves = res.second;
   382	
   383	        if (magnetPriorities == null || magnetPriorities.size() == 0) {
   384	            fileTree.select(true, false);
   385	
   386	        } else {
   387	            /* Select files that have non-IGNORE priority (see BEP35 standard) */
   388	            for (int i = 0; i < files.size(); i++) {
   389	                BencodeFileTree file = treeLeaves[i];
   390	                if (file != null) {
   391	                    Priority p = (i >= magnetPriorities.size() ?
   392	                            Priority.IGNORE :
   393	                            magnetPriorities.get(i));
   394	                    file.select(p != Priority.IGNORE, false);
   395	                }
   396	            }
   397	        }
   398	
   399	        /* Is assigned the root dir of the file tree */
   400	        updateCurDir(fileTree);
   401	    }

   466	    public boolean addTorrent() throws Exception {
   467	        TorrentMetaInfo downloadInfo = info.get();
   468	        if (downloadInfo == null)
   469	            return false;
   470	
   471	        boolean fromMagnet = mutableParams.isFromMagnet();
   472	        DecodeState state = decodeState.getValue();
   473	
   474	        String source = mutableParams.getSource();
   475	        if (source == null)
   476	            return false;
   477	
   478	        Uri dirPath = mutableParams.getDirPath().get();
   479	        if (dirPath == null)
   480	            return false;
   481	
   482	        String name = mutableParams.getName();
   483	        if (TextUtils.isEmpty(name))
   484	            return false;
   485	
   486	        boolean ignoreFreeSpace = mutableParams.isIgnoreFreeSpace();
   487	        Set<Integer> selectedFiles = getSelectedFileIndexes();
   488	        if (!ignoreFreeSpace && state != null &&
   489	                (state.status == Status.DECODE_TORRENT_COMPLETED ||
   490	                        state.status == Status.FETCHING_MAGNET_COMPLETED ||
   491	                        state.status == Status.FETCHING_HTTP_COMPLETED)) {
   492	            if (selectedFiles.isEmpty())
   493	                throw new NoFilesSelectedException();
   494	
   495	            if (!checkFreeSpace())
   496	                throw new FreeSpaceException();
   497	        }
   498	
   499	        Priority[] priorities = new Priority[downloadInfo.fileCount];
   500	        if (downloadInfo.fileCount != 0) {
   501	            if (selectedFiles.size() == downloadInfo.fileCount) {
   502	                Arrays.fill(priorities, Priority.DEFAULT);
   503	            } else {
   504	                Arrays.fill(priorities, Priority.IGNORE);
   505	                for (int index : selectedFiles)
   506	                    priorities[index] = Priority.DEFAULT;
   507	            }
   508	        }
   509	
   510	        AddTorrentParams params = new AddTorrentParams(
   511	                source,
   512	                fromMagnet,
   513	                downloadInfo.sha1Hash,
   514	                name,
   515	                priorities,
   516	                dirPath,
   517	                mutableParams.isSequentialDownload(),
   518	                !mutableParams.isStartAfterAdd(),
   519	                tags,
   520	                mutableParams.isFirstLastPiecePriority()
   521	        );
   522	
   523	        /* TODO: maybe rewrite to WorkManager */
   524	        /* Sync wait inserting */
   525	        Exception[] err = new Exception[1];
   526	        try {
   527	            Thread t = new Thread(() -> {
   528	                try {
   529	                    engine.addTorrentSync(params, false);
   530	
   531	                } catch (Exception e) {
   532	                    err[0] = e;
   533	                }
   534	            });
   535	            t.start();
   536	            t.join();
   537	
   538	        } catch (InterruptedException e) {
   539	            return false;
   540	        }
   541	
   542	        if (err[0] != null)
   543	            throw err[0];
   544	
   545	        return true;
   546	    }
   547	
   548	    private boolean checkFreeSpace() {
   549	        if (fileTree == null)
   550	            return false;
   551	
   552	        long storageFreeSpace = mutableParams.getStorageFreeSpace();
   553	        long treeFreeSpace = fileTree.selectedFileSize();
   554	
   555	        return storageFreeSpace == -1 || storageFreeSpace >= treeFreeSpace;
   556	    }

   565	    private void cancelFetchMagnet() {
   566	        TorrentMetaInfo infoVal = info.get();
   567	        if (infoVal == null)
   568	            return;
   569	
   570	        DecodeState state = decodeState.getValue();
   571	        if (state != null && state.status == Status.FETCHING_MAGNET)
   572	            engine.cancelFetchMagnet(infoVal.sha1Hash);
   573	    }
