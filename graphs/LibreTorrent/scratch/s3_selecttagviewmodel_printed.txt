    20	package org.proninyaroslav.libretorrent.ui.tag;
    21	
    22	import android.app.Application;
    23	
    24	import androidx.annotation.NonNull;
    25	import androidx.lifecycle.AndroidViewModel;
    26	
    27	import org.proninyaroslav.libretorrent.core.RepositoryHelper;
    28	import org.proninyaroslav.libretorrent.core.model.data.entity.TagInfo;
    29	import org.proninyaroslav.libretorrent.core.storage.TagRepository;
    30	
    31	import java.util.HashSet;
    32	import java.util.List;
    33	import java.util.Set;
    34	
    35	import io.reactivex.Flowable;
    36	
    37	public class SelectTagViewModel extends AndroidViewModel {
    38	    private TagRepository tagRepo;
    39	    private Set<Long> excludeTagsId;
    40	
    41	    public SelectTagViewModel(@NonNull Application application) {
    42	        super(application);
    43	
    44	        tagRepo = RepositoryHelper.getTagRepository(application);
    45	    }
    46	
    47	    public void setExcludeTagsId(long[] tagsId) {
    48	        if (tagsId == null) {
    49	            excludeTagsId = null;
    50	        } else {
    51	            excludeTagsId = new HashSet<>();
    52	            for (long tagId : tagsId) {
    53	                excludeTagsId.add(tagId);
    54	            }
    55	        }
    56	    }
    57	
    58	    public boolean filterExcludeTags(@NonNull TagInfo info) {
    59	        return excludeTagsId == null || !excludeTagsId.contains(info.id);
    60	    }
    61	
    62	    Flowable<List<TagInfo>> observeTags() {
    63	        return tagRepo.observeAll();
    64	    }
    65	}
