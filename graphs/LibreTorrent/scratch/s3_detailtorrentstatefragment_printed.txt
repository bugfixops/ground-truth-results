    20	package org.proninyaroslav.libretorrent.ui.detailtorrent.pages;
    21	
    22	import android.content.Context;
    23	import android.os.Bundle;
    24	import android.view.LayoutInflater;
    25	import android.view.View;
    26	import android.view.ViewGroup;
    27	
    28	import androidx.annotation.NonNull;
    29	import androidx.annotation.Nullable;
    30	import androidx.appcompat.app.AppCompatActivity;
    31	import androidx.databinding.DataBindingUtil;
    32	import androidx.fragment.app.Fragment;
    33	import androidx.lifecycle.ViewModelProvider;
    34	
    35	import org.proninyaroslav.libretorrent.R;
    36	import org.proninyaroslav.libretorrent.databinding.FragmentDetailTorrentStateBinding;
    37	import org.proninyaroslav.libretorrent.ui.detailtorrent.DetailTorrentViewModel;
    38	
    39	/*
    40	 * The fragment for displaying torrent state. Part of DetailTorrentFragment.
    41	 */
    42	
    43	public class DetailTorrentStateFragment extends Fragment
    44	{
    45	    private static final String TAG = DetailTorrentStateFragment.class.getSimpleName();
    46	
    47	    private AppCompatActivity activity;
    48	    private DetailTorrentViewModel viewModel;
    49	    private FragmentDetailTorrentStateBinding binding;
    50	
    51	    public static DetailTorrentStateFragment newInstance()
    52	    {
    53	        DetailTorrentStateFragment fragment = new DetailTorrentStateFragment();
    54	
    55	        Bundle b = new Bundle();
    56	        fragment.setArguments(b);
    57	
    58	        return fragment;
    59	    }
    60	
    61	    @Override
    62	    public void onAttach(@NonNull Context context)
    63	    {
    64	        super.onAttach(context);
    65	
    66	        if (context instanceof AppCompatActivity)
    67	            activity = (AppCompatActivity)context;
    68	    }
    69	
    70	    @Override
    71	    public View onCreateView(@NonNull LayoutInflater inflater, ViewGroup container, Bundle savedInstanceState)
    72	    {
    73	        binding = DataBindingUtil.inflate(inflater, R.layout.fragment_detail_torrent_state, container, false);
    74	
    75	        return binding.getRoot();
    76	    }
    77	
    78	    @Override
    79	    public void onViewCreated(@NonNull View view, @Nullable Bundle savedInstanceState) {
    80	        super.onViewCreated(view, savedInstanceState);
    81	
    82	        if (activity == null)
    83	            activity = (AppCompatActivity)getActivity();
    84	
    85	        viewModel = new ViewModelProvider(activity).get(DetailTorrentViewModel.class);
    86	        binding.setViewModel(viewModel);
    87	    }
    88	}
