   356	    private void handleTorrentInfo(Pair<Torrent, TorrentInfo> info)
   357	    {
   358	        Torrent torrent = info.first;
   359	        TorrentInfo ti = info.second;
   360	        TorrentInfo oldTi = viewModel.info.getTorrentInfo();
   361	        boolean alreadyPaused = oldTi != null && oldTi.stateCode == TorrentStateCode.PAUSED;
   362	
   363	        viewModel.updateInfo(torrent, ti);
   364	
   365	        if (ti == null || torrent == null)
   366	            return;
   367	
   368	        if (downloadingMetadata && ti.stateCode != TorrentStateCode.DOWNLOADING_METADATA)
   369	            activity.invalidateOptionsMenu();
   370	        downloadingMetadata = torrent.downloadingMetadata;
   371	
   372	        if (ti.stateCode == TorrentStateCode.PAUSED || alreadyPaused) {
   373	            /* Redraw pause/resume menu */
   374	            if (Utils.isTwoPane(activity))
   375	                prepareOptionsMenu(binding.appbar.toolbar.getMenu());
   376	            else
   377	                activity.invalidateOptionsMenu();
   378	        }
   379	    }
