   237	    private void handleAlertDialogEvent(BaseAlertDialog.Event event) {
   238	        if (event.dialogTag == null) {
   239	            return;
   240	        }
   241	        if (event.dialogTag.equals(TAG_ERR_REPORT_DIALOG)) {
   242	            switch (event.type) {
   243	                case POSITIVE_BUTTON_CLICKED:
   244	                    if (errReportDialog != null) {
   245	                        Dialog dialog = errReportDialog.getDialog();
   246	                        if (dialog != null) {
   247	                            TextInputEditText editText = dialog.findViewById(R.id.comment);
   248	                            Editable e = editText.getText();
   249	                            String comment = (e == null ? null : e.toString());
   250	
   251	                            Utils.reportError(viewModel.errorReport, comment);
   252	                            errReportDialog.dismiss();
   253	                        }
   254	                    }
   255	                    finish();
   256	                    break;
   257	                case NEGATIVE_BUTTON_CLICKED:
   258	                    if (errReportDialog != null) {
   259	                        errReportDialog.dismiss();

   262	                    break;
   263	            }
   264	        } else if (event.dialogTag.equals(TAG_PERM_DENIED_DIALOG)) {
   265	            if (event.type != BaseAlertDialog.EventType.DIALOG_SHOWN) {
   266	                permDeniedDialog.dismiss();
   267	            }
   268	            if (event.type == BaseAlertDialog.EventType.NEGATIVE_BUTTON_CLICKED) {
   269	                permissionManager.requestPermissions();
   270	            }
   271	            if (event.type == BaseAlertDialog.EventType.POSITIVE_BUTTON_CLICKED) {
   272	                permissionManager.setDoNotAskStorage(true);

   305	    private void observeDecodeState()
   306	    {
   307	        viewModel.getDecodeState().observe(this, (state) -> {
   308	            switch (state.status) {
   309	                case UNKNOWN:
   310	                    Uri uri = getUri();
   311	                    if (uri != null)
   312	                        viewModel.startDecode(uri);
   313	                    break;
   314	                case DECODE_TORRENT_FILE:
   315	                case FETCHING_HTTP:
   316	                case FETCHING_MAGNET:
   317	                    onStartDecode(state.status == AddTorrentViewModel.Status.DECODE_TORRENT_FILE);
   318	                    break;
   319	                case FETCHING_HTTP_COMPLETED:
   320	                case DECODE_TORRENT_COMPLETED:
   321	                case FETCHING_MAGNET_COMPLETED:
   322	                case ERROR:
   323	                    onStopDecode(state.error);
   324	                    break;
   325	            }
   326	        });
   327	    }

   391	    private void handleAddException(Throwable e)
   392	    {
   393	        if (e instanceof NoFilesSelectedException) {
   394	            Snackbar.make(binding.coordinatorLayout,
   395	                    R.string.error_no_files_selected,
   396	                    Snackbar.LENGTH_LONG)
   397	                    .show();
   398	            return;
   399	        }
   400	
   401	        if (e instanceof FreeSpaceException) {
   402	            Snackbar.make(binding.coordinatorLayout,
   403	                    R.string.error_free_space,
   404	                    Snackbar.LENGTH_LONG)
   405	                    .show();
   406	            return;
   407	        }
   408	
   409	        Log.e(TAG, Log.getStackTraceString(e));
   410	        if (e instanceof FileNotFoundException) {
   411	            showAddErrorDialog(getApplication().getString(R.string.error_file_not_found_add_torrent), null);
   412	        } else if (e instanceof IOException) {
   413	            showAddErrorDialog(getApplication().getString(R.string.error_io_add_torrent), null);
   414	        } else {
   415	            showAddErrorDialog(getApplication().getString(R.string.error_add_torrent), e);
   416	        }
