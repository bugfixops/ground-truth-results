  436     public static void enableBootReceiver(@NonNull Context context, boolean enable) {
  437         SettingsRepository pref = RepositoryHelper.getSettingsRepository(context);
  438         boolean schedulingStart = pref.enableSchedulingStart();
  439         boolean schedulingStop = pref.enableSchedulingShutdown();
  440         boolean autostart = pref.autostart();
  441         int flag = (!(enable || schedulingStart || schedulingStop || autostart) ?
  442                 PackageManager.COMPONENT_ENABLED_STATE_DISABLED :
  443                 PackageManager.COMPONENT_ENABLED_STATE_ENABLED);
  444         ComponentName bootReceiver = new ComponentName(context, BootReceiver.class);
  445         context.getPackageManager()
  446                 .setComponentEnabledSetting(bootReceiver, flag, PackageManager.DONT_KILL_APP);
