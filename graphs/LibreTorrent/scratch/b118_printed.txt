   129	    private void subscribeAlertDialog()
   130	    {
   131	        Disposable d = dialogViewModel.observeEvents()
   132	                .subscribe((event) -> {
   133	                    if (event.dialogTag == null || !event.dialogTag.equals(TAG_EXACT_ALARM_PERMISSION_DIALOG)) {
   134	                        return;
   135	                    }
   136	                    if (event.type == BaseAlertDialog.EventType.POSITIVE_BUTTON_CLICKED) {
   137	                        Utils.requestExactAlarmPermission(getContext());
   138	                    }
   139	                });
   140	        disposables.add(d);
   141	    }
   142	
   143	    private void showExactAlarmPermissionDialog()
   144	    {
   145	        if (!isAdded())
   146	            return;
   147	
   148	        var fm = getChildFragmentManager();
   149	        if (fm.findFragmentByTag(TAG_EXACT_ALARM_PERMISSION_DIALOG) == null) {
   150	            var exactAlarmDialog = BaseAlertDialog.newInstance(
   151	                    getString(R.string.permission_denied),
   152	                    getString(R.string.exact_alarm_permission_warning),
   153	                    0,
   154	                    getString(R.string.yes),
   155	                    getString(R.string.no),
   156	                    null,
   157	                    true);
   158	
   159	            exactAlarmDialog.show(fm, TAG_EXACT_ALARM_PERMISSION_DIALOG);
   160	        }
   161	    }

   175	    public void onDisplayPreferenceDialog(Preference preference)
   176	    {
   177	        DialogFragment dialogFragment = null;
   178	        if (preference instanceof TimePreference) {
   179	            dialogFragment = TimePreferenceDialogFragmentCompat.newInstance(preference.getKey());
   180	        }
   181	        if (dialogFragment != null && isAdded()) {
   182	            dialogFragment.setTargetFragment(this, 0);
   183	            dialogFragment.show(this.getParentFragmentManager(), "android.support.v7.preference" +
   184	                    ".PreferenceFragment.DIALOG");
   185	        } else {
   186	            super.onDisplayPreferenceDialog(preference);
   187	        }
   188	    }

   190	    @Override
   191	    public boolean onPreferenceChange(Preference preference, Object newValue)
   192	    {
   193	        Context context = getActivity().getApplicationContext();
   194	
   195	        if (preference.getKey().equals(getString(R.string.pref_key_enable_scheduling_start))) {
   196	            pref.enableSchedulingStart((boolean)newValue);
   197	
   198	            if ((boolean)newValue) {
   199	                int time = pref.schedulingStartTime();
   200	                if (!Scheduler.setStartAppAlarm(context, time)) {
   201	                    showExactAlarmPermissionDialog();
   202	                }
   203	            } else {
   204	                Scheduler.cancelStartAppAlarm(context);
   205	            }
   206	            Utils.enableBootReceiver(context, (boolean)newValue);
   207	
   208	        } else if (preference.getKey().equals(getString(R.string.pref_key_enable_scheduling_shutdown))) {
   209	            pref.enableSchedulingShutdown((boolean)newValue);
   210	
   211	            if ((boolean)newValue) {
   212	                int time = pref.schedulingStartTime();
   213	                if (!Scheduler.setStopAppAlarm(context, time)) {
   214	                    showExactAlarmPermissionDialog();
   215	                }
   216	            } else {
   217	                Scheduler.cancelStopAppAlarm(context);
   218	            }
   219	            Utils.enableBootReceiver(getActivity(), (boolean)newValue);
   220	
   221	        } else if (preference.getKey().equals(getString(R.string.pref_key_scheduling_start_time))) {
   222	            pref.schedulingStartTime((int)newValue);
   223	            if (!Scheduler.setStartAppAlarm(context, (int)newValue)) {
   224	                showExactAlarmPermissionDialog();
   225	            }
   226	
   227	        } else if (preference.getKey().equals(getString(R.string.pref_key_scheduling_shutdown_time))) {
   228	            pref.schedulingShutdownTime((int)newValue);
   229	            if (!Scheduler.setStopAppAlarm(context, (int)newValue)) {
   230	                showExactAlarmPermissionDialog();
   231	            }
   232	
   233	        } else if (preference.getKey().equals(getString(R.string.pref_key_scheduling_run_only_once))) {
   234	            pref.schedulingRunOnlyOnce((boolean)newValue);
   235	
   236	        } else if (preference.getKey().equals(getString(R.string.pref_key_scheduling_switch_wifi))) {
   237	            pref.schedulingSwitchWiFi((boolean)newValue);
   238	        }
   239	
   240	        return true;
   241	    }

   227	    private void dirChooseDialog(Uri path)
   228	    {
   229	        String dirPath = null;
   230	        if (path != null && Utils.isFileSystemPath(path))
   231	            dirPath = path.getPath();
   232	
   233	        Intent i = new Intent(getActivity(), FileManagerDialog.class);
   234	        FileManagerConfig config = new FileManagerConfig(dirPath,
   235	                null,
   236	                FileManagerConfig.DIR_CHOOSER_MODE);
   237	        i.putExtra(FileManagerDialog.TAG_CONFIG, config);
   238	
   239	        downloadDirChoose.launch(i);
   240	    }
   241	    final ActivityResultLauncher<Intent> downloadDirChoose = registerForActivityResult(
   242	            new ActivityResultContracts.StartActivityForResult(),
   243	            result -> {
   244	                Intent data = result.getData();
   245	                if (result.getResultCode() != Activity.RESULT_OK) {
   246	                   return;
   247	                }
   248	                if (data.getData() == null || dirChooserBindPref == null)
   249	                    return;
   250	
   251	                Uri path = data.getData();
   252	
   253	                Preference p = findPreference(dirChooserBindPref);
   254	                if (p == null)
   255	                    return;
   256	
   257	                if (dirChooserBindPref.equals(getString(R.string.pref_key_dir_to_watch))) {
   258	                    pref.dirToWatch(path.toString());
   259	
   260	                } else if (dirChooserBindPref.equals(getString(R.string.pref_key_move_after_download_in))) {
   261	                    pref.moveAfterDownloadIn(path.toString());
   262	
   263	                } else if (dirChooserBindPref.equals(getString(R.string.pref_key_save_torrent_files_in))) {
   264	                    pref.saveTorrentFilesIn(path.toString());
   265	
   266	                } else if (dirChooserBindPref.equals(getString(R.string.pref_key_save_torrents_in))) {
   267	                    pref.saveTorrentsIn(path.toString());
   268	                }
   269	
   270	                try {
   271	                    p.setSummary(fs.getDirPath(path));
   272	                } catch (UnknownUriException e) {
   273	                    Log.e(TAG, Log.getStackTraceString(e));
   274	                }
   275	            }
   276	    );

    24	import android.content.DialogInterface;
    25	import android.content.Intent;
    26	import android.os.Bundle;
    27	import android.view.KeyEvent;
    28	import android.view.LayoutInflater;
    29	import android.view.View;
    30	import android.widget.Button;
    31	
    32	import androidx.annotation.NonNull;
    33	import androidx.annotation.Nullable;
    34	import androidx.appcompat.app.AlertDialog;
    35	import androidx.appcompat.app.AppCompatActivity;
    36	import androidx.databinding.DataBindingUtil;
    37	import androidx.fragment.app.DialogFragment;
    38	import androidx.lifecycle.ViewModelProvider;
    39	import androidx.recyclerview.widget.LinearLayoutManager;
    40	
    41	import org.proninyaroslav.libretorrent.R;
    42	import org.proninyaroslav.libretorrent.databinding.DialogSelectTagBinding;
    43	import org.proninyaroslav.libretorrent.ui.FragmentCallback;
    44	import org.proninyaroslav.libretorrent.ui.addtag.AddTagActivity;
    45	
    46	import io.reactivex.Flowable;
    47	import io.reactivex.android.schedulers.AndroidSchedulers;
    48	import io.reactivex.disposables.CompositeDisposable;
    49	import io.reactivex.schedulers.Schedulers;
    50	
    51	public class SelectTagDialog extends DialogFragment implements TagsAdapter.OnClickListener {
    52	    private static final String TAG = SelectTagDialog.class.getSimpleName();
    53	
    54	    private static final String TAG_EXCLUDE_TAGS_ID = "exclude_tags_id";
    55	
    56	    private AlertDialog alert;
    57	    private AppCompatActivity activity;
    58	    private SelectTagViewModel viewModel;
    59	    private TagsAdapter adapter;
    60	    private DialogSelectTagBinding binding;
    61	    private final CompositeDisposable disposables = new CompositeDisposable();
    62	
    63	    public static SelectTagDialog newInstance(@Nullable long[] excludeTagsId) {
    64	        SelectTagDialog frag = new SelectTagDialog();
    65	
    66	        Bundle args = new Bundle();
    67	        args.putLongArray(TAG_EXCLUDE_TAGS_ID, excludeTagsId);
    68	        frag.setArguments(args);
    69	
    70	        return frag;
    71	    }
    72	
    73	    @Override
    74	    public void onAttach(@NonNull Context context) {
    75	        super.onAttach(context);
    76	
    77	        if (context instanceof AppCompatActivity) {
    78	            activity = (AppCompatActivity) context;
    79	        }
    80	    }
    81	
    82	    @Override
    83	    public void onResume() {
    84	        super.onResume();
    85	
    86	        // Back button handle
    87	        getDialog().setOnKeyListener((dialog, keyCode, event) -> {
    88	            if (keyCode == KeyEvent.KEYCODE_BACK) {
    89	                if (event.getAction() == KeyEvent.ACTION_DOWN) {
    90	                    onBackPressed();
    91	                }
    92	                return true;
    93	            } else {
    94	                return false;
    95	            }
    96	        });
    97	    }
    98	
    99	    @Override
   100	    public void onStart() {
   101	        super.onStart();
   102	
   103	        subscribeTags();
   104	    }
   105	
   106	    @Override
   107	    public void onStop() {
   108	        super.onStop();
   109	
   110	        disposables.clear();
   111	    }
   112	
   113	    @NonNull
   114	    @Override
   115	    public Dialog onCreateDialog(Bundle savedInstanceState) {
   116	        if (activity == null) {
   117	            activity = (AppCompatActivity) getActivity();
   118	        }
   119	
   120	        ViewModelProvider provider = new ViewModelProvider(activity);
   121	        viewModel = provider.get(SelectTagViewModel.class);
   122	        if (getArguments() != null) {
   123	            viewModel.setExcludeTagsId(getArguments()
   124	                    .getLongArray(TAG_EXCLUDE_TAGS_ID));
   125	        }
   126	
   127	        LayoutInflater i = LayoutInflater.from(activity);
   128	        binding = DataBindingUtil.inflate(i, R.layout.dialog_select_tag, null, false);
   129	
   130	        adapter = new TagsAdapter(this);
   131	
   132	        initLayoutView();
   133	        initAlertDialog(binding.getRoot());
   134	
   135	        return alert;
   136	    }
   137	
   138	    private void initLayoutView() {
   139	        binding.tagsList.setEmptyView(binding.emptyListView);
   140	        binding.tagsList.setLayoutManager(new LinearLayoutManager(activity));
   141	        binding.tagsList.setAdapter(adapter);
   142	    }
   143	
   144	    private void initAlertDialog(View view) {
   145	        AlertDialog.Builder builder = new AlertDialog.Builder(activity)
   146	                .setTitle(R.string.select_tag)
   147	                .setPositiveButton(R.string.new_tag, null)
   148	                .setNegativeButton(R.string.cancel, null)
   149	                .setView(view);
   150	
   151	        alert = builder.create();
   152	        alert.setCanceledOnTouchOutside(false);
   153	        alert.setOnShowListener((DialogInterface dialog) -> {
   154	            Button newTagButton = alert.getButton(AlertDialog.BUTTON_POSITIVE);
   155	            Button cancelButton = alert.getButton(AlertDialog.BUTTON_NEGATIVE);
   156	            newTagButton.setOnClickListener(
   157	                    (v) -> startActivity(new Intent(activity, AddTagActivity.class))
   158	            );
   159	            cancelButton.setOnClickListener(
   160	                    (v) -> finish(new Intent(), FragmentCallback.ResultCode.CANCEL)
   161	            );
   162	        });
   163	    }
   164	
   165	    private void subscribeTags() {
   166	        disposables.add(viewModel.observeTags()
   167	                .subscribeOn(Schedulers.io())
   168	                .observeOn(AndroidSchedulers.mainThread())
   169	                .flatMapSingle((list) -> Flowable.fromIterable(list)
   170	                                .filter(viewModel::filterExcludeTags)
   171	                                .map(TagItem::new)
   172	                                .toList()
   173	                        )
   174	                        .subscribe(adapter::submitList)
   175	                );
   176	    }
   177	
   178	    @Override
   179	    public void onTagClicked(@NonNull TagItem item) {
   180	        Intent i = new Intent();
   181	        i.putExtra(SelectTagActivity.TAG_RESULT_SELECTED_TAG, item.info);
   182	
   183	        finish(i, FragmentCallback.ResultCode.OK);
   184	    }
   185	
   186	    public void onBackPressed() {
   187	
   188	        finish(new Intent(), FragmentCallback.ResultCode.BACK);
   189	    }
   190	
   191	    private void finish(Intent intent, FragmentCallback.ResultCode code) {
   192	        alert.dismiss();
   193	        ((FragmentCallback) activity).onFragmentFinished(this, intent, code);
   194	    }
   195	}
