   251	    private void subscribeTorrentInfo()
   252	    {
   253	        if (torrentId == null)
   254	            return;
   255	
   256	        disposables.add(viewModel.observeTorrentMetaInfo()
   257	                .subscribeOn(Schedulers.io())
   258	                .observeOn(AndroidSchedulers.mainThread())
   259	                .subscribe((metaInfo) -> viewModel.updateInfo(metaInfo),
   260	                        (Throwable e) -> {
   261	                            Log.e(TAG, "Getting meta info error: " + Log.getStackTraceString(e));
   262	                            finish(new Intent(), FragmentCallback.ResultCode.CANCEL);
   263	                        }));
   264	
   265	        disposables.add(viewModel.observeTorrentInfoPair()
   266	                .subscribeOn(Schedulers.io())
   267	                .observeOn(AndroidSchedulers.mainThread())
   268	                .subscribe(this::handleTorrentInfo,
   269	                        (Throwable e) -> Log.e(TAG, "Getting info error: " + Log.getStackTraceString(e))));
   270	
   271	        disposables.add(viewModel.observeAdvancedTorrentInfo()
   272	                .subscribeOn(Schedulers.io())
   273	                .observeOn(AndroidSchedulers.mainThread())
   274	                .subscribe((info) -> viewModel.updateInfo(info),
   275	                        (Throwable e) -> Log.e(TAG, "Getting advanced info error: " + Log.getStackTraceString(e))));
   276	    }

   297	    private void subscribeAlertDialog()
   298	    {
   299	        Disposable d = dialogViewModel.observeEvents()
   300	                .subscribe((event) -> {
   301	                    if (event.dialogTag == null)
   302	                        return;
   303	
   304	                    switch (event.type) {
   305	                        case POSITIVE_BUTTON_CLICKED:
   306	                            if (event.dialogTag.equals(TAG_DELETE_TORRENT_DIALOG) && deleteTorrentDialog != null) {
   307	                                deleteTorrent();
   308	                                deleteTorrentDialog.dismiss();
   309	                            } else if (event.dialogTag.equals(TAG_MAGNET_INCLUDE_PRIOR_DIALOG)) {
   310	                                shareMagnet(true);
   311	                            } else if (event.dialogTag.equals(TAG_ERR_REPORT_DIALOG) && errReportDialog != null) {
   312	                                Dialog dialog = errReportDialog.getDialog();
   313	                                if (dialog != null) {
   314	                                    TextInputEditText editText = dialog.findViewById(R.id.comment);
   315	                                    Editable e = editText.getText();
   316	                                    String comment = (e == null ? null : e.toString());
   317	
   318	                                    Utils.reportError(viewModel.errorReport, comment);
   319	                                    errReportDialog.dismiss();
   320	                                }
   321	                            } else if (event.dialogTag.equals(TAG_ADD_TRACKERS_DIALOG) && addTrackersDialog != null) {
   322	                                addTrackers(false);
   323	                            } else if (event.dialogTag.equals(TAG_SPEED_LIMIT_DIALOG) && speedLimitDialog != null) {
   324	                                setSpeedLimit();
   325	                                speedLimitDialog.dismiss();
   326	                            }
   327	                            break;
   328	                        case NEGATIVE_BUTTON_CLICKED:
   329	                            if (event.dialogTag.equals(TAG_DELETE_TORRENT_DIALOG) && deleteTorrentDialog != null) {
   330	                                deleteTorrentDialog.dismiss();
   331	                            } else if (event.dialogTag.equals(TAG_MAGNET_INCLUDE_PRIOR_DIALOG)) {
   332	                                shareMagnet(false);
   333	                            } else if (event.dialogTag.equals(TAG_ERR_REPORT_DIALOG) && errReportDialog != null) {
   334	                                errReportDialog.dismiss();
   335	                            } else if (event.dialogTag.equals(TAG_ADD_TRACKERS_DIALOG) && addTrackersDialog != null) {
   336	                                addTrackers(true);
   337	                            } else if (event.dialogTag.equals(TAG_SPEED_LIMIT_DIALOG) && speedLimitDialog != null) {
   338	                                speedLimitDialog.dismiss();
   339	                            }
   340	                            break;
   341	                        case NEUTRAL_BUTTON_CLICKED:
   342	                            if (event.dialogTag.equals(TAG_ADD_TRACKERS_DIALOG) && addTrackersDialog != null)
   343	                                addTrackersDialog.dismiss();
   344	                            break;
   345	                        case DIALOG_SHOWN:
   346	                            if (event.dialogTag.equals(TAG_ADD_TRACKERS_DIALOG) && addTrackersDialog != null)
   347	                                initAddTrackersDialog();
   348	                            else if (event.dialogTag.equals(TAG_SPEED_LIMIT_DIALOG) && speedLimitDialog != null)
   349	                                initSpeedLimitDialog();
   350	                            break;
   351	                    }
   352	                });
   353	        disposables.add(d);
   354	    }

   356	    private void handleTorrentInfo(Pair<Torrent, TorrentInfo> info)
   357	    {
   358	        Torrent torrent = info.first;
   359	        TorrentInfo ti = info.second;
   360	        TorrentInfo oldTi = viewModel.info.getTorrentInfo();
   361	        boolean alreadyPaused = oldTi != null && oldTi.stateCode == TorrentStateCode.PAUSED;
   362	
   363	        viewModel.updateInfo(torrent, ti);
   364	
   365	        if (ti == null || torrent == null)
   366	            return;
   367	
   368	        if (downloadingMetadata && ti.stateCode != TorrentStateCode.DOWNLOADING_METADATA)
   369	            activity.invalidateOptionsMenu();
   370	        downloadingMetadata = torrent.downloadingMetadata;
   371	
   372	        if (ti.stateCode == TorrentStateCode.PAUSED || alreadyPaused) {
   373	            /* Redraw pause/resume menu */
   374	            if (Utils.isTwoPane(activity))
   375	                prepareOptionsMenu(binding.appbar.toolbar.getMenu());
   376	            else
   377	                activity.invalidateOptionsMenu();
   378	        }
   379	    }

   428	    private void prepareOptionsMenu(Menu menu)
   429	    {
   430	        MenuItem pauseResume = menu.findItem(R.id.pause_resume_torrent_menu);
   431	        Torrent torrent = viewModel.info.getTorrent();
   432	        TorrentInfo ti = viewModel.info.getTorrentInfo();
   433	        if (torrent == null || ti == null || ti.stateCode != TorrentStateCode.PAUSED) {
   434	            pauseResume.setTitle(R.string.pause_torrent);
   435	            if (!Utils.isTwoPane(activity))
   436	                pauseResume.setIcon(R.drawable.ic_pause_white_24dp);
   437	
   438	        } else {
   439	            pauseResume.setTitle(R.string.resume_torrent);
   440	            if (!Utils.isTwoPane(activity))
   441	                pauseResume.setIcon(R.drawable.ic_play_arrow_white_24dp);
   442	        }
   443	
   444	        MenuItem saveTorrentFile = menu.findItem(R.id.save_torrent_file_menu);
   445	        if (torrent != null && torrent.downloadingMetadata)
   446	            saveTorrentFile.setVisible(false);
   447	        else
   448	            saveTorrentFile.setVisible(true);
   449	    }

   451	    @Override
   452	    public boolean onOptionsItemSelected(@NonNull MenuItem item)
   453	    {
   454	        int itemId = item.getItemId();
   455	        if (itemId == android.R.id.home) {
   456	            onBackPressed();
   457	        } else if (itemId == R.id.pause_resume_torrent_menu) {
   458	            viewModel.pauseResumeTorrent();
   459	        } else if (itemId == R.id.delete_torrent_menu) {
   460	            deleteTorrentDialog();
   461	        } else if (itemId == R.id.force_recheck_torrent_menu) {
   462	            viewModel.forceRecheckTorrent();
   463	        } else if (itemId == R.id.force_announce_torrent_menu) {
   464	            viewModel.forceAnnounceTorrent();
   465	        } else if (itemId == R.id.share_magnet_menu) {
   466	            shareMagnetDialog();
   467	        } else if (itemId == R.id.save_torrent_file_menu) {
   468	            torrentSaveChooseDialog();
   469	        } else if (itemId == R.id.add_trackers_menu) {
   470	            addTrackersDialog();
   471	        } else if (itemId == R.id.torrent_speed_limit) {
   472	            speedLimitDialog();
   473	        }
   474	
   475	        return true;
   476	    }

   548	    private void saveErrorTorrentFileDialog(Exception e)
   549	    {
   550	        if (!isAdded())
   551	            return;
   552	
   553	        viewModel.errorReport = e;
   554

   644	    private boolean checkAddTrackersField(List<String> strings,
   645	                                          TextInputLayout layout,
   646	                                          TextInputEditText field)
   647	    {
   648	        if (strings == null || layout == null)
   649	            return false;
   650	
   651	        if (strings.isEmpty()) {
   652	            layout.setErrorEnabled(true);
   653	            layout.setError(getString(R.string.error_empty_link));
   654	            layout.requestFocus();
   655	
   656	            return false;
   657	        }
   658	
   659	        boolean valid = true;
   660	        int curLineStartIndex = 0;
   661	        for (String s : strings) {
   662	            if (!Utils.isValidTrackerUrl(s) && field.getText() != null) {
   663	                TypedArray a = activity.obtainStyledAttributes(new TypedValue().data, new int[]{ R.attr.colorError});
   664	                /* Select invalid url */
   665	                field.getText().setSpan(new ForegroundColorSpan(a.getColor(0, 0)),
   666	                        curLineStartIndex,
   667	                        curLineStartIndex + s.length(),
   668	                        Spanned.SPAN_EXCLUSIVE_EXCLUSIVE);
   669	                a.recycle();
   670	
   671	                valid = false;
   672	            }
   673	            /* Considering newline char */
   674	            curLineStartIndex += s.length() + 1;
   675	        }
   676	
   677	        if (valid) {
   678	            layout.setErrorEnabled(false);
   679	            layout.setError(null);
   680	        } else {
   681	            layout.setErrorEnabled(true);
   682	            layout.setError(getString(R.string.error_invalid_link));
   683	            layout.requestFocus();
   684	        }
   685	
   686	        return valid;
   687	    }

   689	    private void addTrackers(boolean replace)
   690	    {
   691	        Dialog dialog = addTrackersDialog.getDialog();
   692	        if (dialog == null)
   693	            return;
   694	
   695	        TextInputEditText field = dialog.findViewById(R.id.multiline_text_input_dialog);
   696	        TextInputLayout fieldLayout = dialog.findViewById(R.id.layout_multiline_text_input_dialog);
   697	
   698	        Editable editable = field.getText();
   699	        if (editable == null)
   700	            return;
   701	
   702	        String text = editable.toString();
   703	
   704	        List<String> urls = Observable.fromArray(text.split(Utils.getLineSeparator()))
   705	                .filter((s) -> !s.isEmpty())
   706	                .toList()
   707	                .blockingGet();
   708	
   709	        if (checkAddTrackersField(urls, fieldLayout, field))
   710	            addTrackersDialog.dismiss();
   711	        else
   712	            return;
   713	
   714	        if (checkAddTrackersField(urls, fieldLayout, field))
   715	            addTrackersDialog.dismiss();
   716	        else
   717	            return;
   718	
   719	        NormalizeUrl.Options options = new NormalizeUrl.Options();
   720	        options.decode = false;
   721	        List<String> normalizedUrls = new ArrayList<>(urls.size());
   722	        for (String url : urls) {
   723	            if (TextUtils.isEmpty(url))
   724	                continue;
   725	            try {
   726	                normalizedUrls.add(NormalizeUrl.normalize(url, options));
   727	
   728	            } catch (NormalizeUrlException e) {
   729	                /* Ignore */
   730	            }
   731	        }
   732	
   733	        if (replace)
   734	            viewModel.replaceTrackers(normalizedUrls);
   735	        else
   736	            viewModel.addTrackers(normalizedUrls);
   737	    }

   740	    {
   741	        Dialog dialog = speedLimitDialog.getDialog();
   742	        if (dialog == null)

   745	        TextInputEditText upload = dialog.findViewById(R.id.upload_limit);

   747	
   748	        int minSpeedLimit = 0;

   750	
   751	        upload.setFilters(filter);
   752	
   753	        int uploadSpeedLimit = viewModel.getUploadSpeedLimit();
   754	        int downloadSpeedLimit = viewModel.getDownloadSpeedLimit();
   755	
   756	        if (TextUtils.isEmpty(upload.getText()))
   757	            upload.setText((uploadSpeedLimit != -1 ?
   758	                    Integer.toString(uploadSpeedLimit / 1024) : Integer.toString(minSpeedLimit)));
   759	
   760	        download.setFilters(filter);
   761	        if (TextUtils.isEmpty(download.getText()))
   762	            download.setText((downloadSpeedLimit != -1 ?
   763	                    Integer.toString(downloadSpeedLimit / 1024) : Integer.toString(minSpeedLimit)));
   764	    }

   766	    private void setSpeedLimit()
   767	    {
   768	        Dialog dialog = speedLimitDialog.getDialog();
   769	        if (dialog == null)
   770	            return;
   771	
   772	        TextInputEditText upload = dialog.findViewById(R.id.upload_limit);
   773	        TextInputEditText download = dialog.findViewById(R.id.download_limit);
   774	
   775	        Editable uploadEditable = upload.getText();
   776	        Editable downloadEditable = download.getText();
   777	        if (TextUtils.isEmpty(uploadEditable) || TextUtils.isEmpty(uploadEditable))
   778	            return;
   779	
   780	        int uploadSpeedLimit;
   781	        try {
   782	            uploadSpeedLimit = Integer.parseInt(uploadEditable.toString()) * 1024;
   783	        } catch (NumberFormatException e) {
   784	            uploadSpeedLimit = 0;
   785	        }
   786	        int downloadSpeedLimit;
   787	        try {
   788	            downloadSpeedLimit = Integer.parseInt(downloadEditable.toString()) * 1024;
   789	        } catch (NumberFormatException e) {
   790	            downloadSpeedLimit = 0;
   791	        }
   792	        viewModel.setSpeedLimit(uploadSpeedLimit, downloadSpeedLimit);
   793	    }
