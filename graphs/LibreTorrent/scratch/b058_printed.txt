568:         MessageDigest messageDigest;
569:         try {
570:             messageDigest = MessageDigest.getInstance("SHA1");
571:         } catch (NoSuchAlgorithmException e) {
572:             return null;
573:         }
574:         messageDigest.update(s.getBytes(StandardCharsets.UTF_8));
575:         StringBuilder sha1 = new StringBuilder();
576:         for (byte b : messageDigest.digest()) {
577:             if ((0xff & b) < 0x10)
578:                 sha1.append("0");
