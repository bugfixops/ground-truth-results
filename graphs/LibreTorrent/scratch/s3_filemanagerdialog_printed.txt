   281	    private void registerMediaReceiver() {
   282	        IntentFilter filter = new IntentFilter();
   283	        filter.addAction(Intent.ACTION_MEDIA_MOUNTED);
   284	        filter.addAction(Intent.ACTION_MEDIA_UNMOUNTED);
   285	        filter.addDataScheme("file");
   286	        registerReceiver(mediaReceiver, filter);
   287	    }

   319	    private void handleAlertDialogEvent(BaseAlertDialog.Event event) {
   320	        if (event.dialogTag == null)
   321	            return;
   322	
   323	        switch (event.type) {
   324	            case POSITIVE_BUTTON_CLICKED:
   325	                if (event.dialogTag.equals(TAG_INPUT_NAME_DIALOG) && inputNameDialog != null) {
   326	                    Dialog dialog = inputNameDialog.getDialog();
   327	                    if (dialog != null) {
   328	                        EditText nameField = dialog.findViewById(R.id.text_input_dialog);
   329	                        String name = nameField.getText().toString();
   330	
   331	                        if (!viewModel.createDirectory(name)) {
   332	                            showCreateFolderErrDialog();
   333	                        } else {
   334	                            try {
   335	                                viewModel.openDirectory(name);
   336	
   337	                            } catch (SecurityException e) {
   338	                                permissionDeniedToast();
   339	                            } catch (IOException e) {
   340	                                Log.e(TAG, Log.getStackTraceString(e));
   341	                                showSendErrorDialog(e);
   342	                            }
   343	                        }
   344	                    }
   345	                    inputNameDialog.dismiss();
   346	
   347	                } else if (event.dialogTag.equals(TAG_REPLACE_FILE_DIALOG)) {
   348	                    createFile(true);
   349	                } else if (event.dialogTag.equals(TAG_ERROR_REPORT_DIALOG) && errorReportDialog != null) {
   350	                    Dialog dialog = errorReportDialog.getDialog();
   351	                    if (dialog != null) {
   352	                        TextInputEditText editText = dialog.findViewById(R.id.comment);
   353	                        Editable e = editText.getText();
   354	                        String comment = (e == null ? null : e.toString());
   355	
   356	                        Utils.reportError(viewModel.errorReport, comment);
   357	                        errorReportDialog.dismiss();
   358	                    }
   359	                }
   360	                break;
   361	            case NEGATIVE_BUTTON_CLICKED:
   362	                if (event.dialogTag.equals(TAG_INPUT_NAME_DIALOG) && inputNameDialog != null)
   363	                    inputNameDialog.dismiss();
   364	                else if (event.dialogTag.equals(TAG_ERROR_REPORT_DIALOG) && errorReportDialog != null)
   365	                    errorReportDialog.dismiss();
   366	                break;
   367	        }
   368	    }

   429	    @Override
   430	    public void onItemClicked(FileManagerNode item) {
   431	        if (item.getName().equals(FileManagerNode.PARENT_DIR)) {
   432	            try {
   433	                viewModel.upToParentDirectory();
   434	
   435	            } catch (SecurityException e) {
   436	                permissionDeniedToast();
   437	            }
   438	            return;
   439	        }
   440	
   441	        if (item.getType() == FileManagerNode.Type.DIR) {
   442	            try {
   443	                viewModel.openDirectory(item.getName());
   444	
   445	            } catch (SecurityException e) {
   446	                permissionDeniedToast();
   447	            } catch (IOException e) {
   448	                Log.e(TAG, Log.getStackTraceString(e));
   449	                showSendErrorDialog(e);
   450	            }
   451	
   452	        } else if (item.getType() == FileManagerNode.Type.FILE &&
   453	                viewModel.config.showMode == FileManagerConfig.FILE_CHOOSER_MODE) {
   454	            saveCurDirectoryPath();
   455	            returnFileUri(item.getName());
   456	        }
   457	    }

   464	    private void saveCurDirectoryPath() {
   465	        String path = viewModel.curDir.get();
   466	        if (path == null)
   467	            return;
   468	
   469	        String keyFileManagerLastDir = getString(R.string.pref_key_filemanager_last_dir);
   470	        if (!pref.getString(keyFileManagerLastDir, "").equals(path))
   471	            pref.edit().putString(keyFileManagerLastDir, path).apply();
   472	    }

   498	    @Override
   499	    public boolean onOptionsItemSelected(MenuItem item) {
   500	        int itemId = item.getItemId();
   501	        if (itemId == android.R.id.home) {
   502	            onBackPressed();
   503	        } else if (itemId == R.id.filemanager_home_menu) {
   504	            openHomeDirectory();
   505	        } else if (itemId == R.id.filemanager_ok_menu) {
   506	            saveCurDirectoryPath();
   507	            if (viewModel.config.showMode == FileManagerConfig.SAVE_FILE_MODE)
   508	                createFile(false);
   509	            else
   510	                returnDirectoryUri();
   511	        } else if (itemId == R.id.filemanager_go_to_folder_menu) {
   512	            goToDirectoryDialog();
   513	        }
   514	
   515	        return true;
   516	    }

   524	    private void goToDirectory(String path) {
   525	        if (!TextUtils.isEmpty(path)) {
   526	            try {
   527	                viewModel.jumpToDirectory(path);
   528	
   529	            } catch (SecurityException e) {
   530	                permissionDeniedToast();
   531	            }
   532	        } else {
   533	            if (getSupportFragmentManager().findFragmentByTag(TAG_GO_TO_DIR_DIALOG) == null) {
   534	                BaseAlertDialog errDialog = BaseAlertDialog.newInstance(
   535	                        getString(R.string.go_to_folder),
   536	                        getString(R.string.error_open_dir),
   537	                        0,
   538	                        getString(R.string.ok),
   539	                        null,
   540	                        null,
   541	                        true);
   542	
   543	                errDialog.show(getSupportFragmentManager(), TAG_GO_TO_DIR_DIALOG);
   544	            }
   545	        }
   546	    }

   557	    private void returnDirectoryUri() {
   558	        Intent i = new Intent();
   559	        try {
   560	            i.setData(viewModel.getCurDirectoryUri());
   561	
   562	        } catch (SecurityException e) {

   565	        }
   566	        setResult(RESULT_OK, i);
   567	        finish();

   570	    private void createFile(boolean replace) {
   571	        if (!checkFileNameField())
   572	            return;
   573	
   574	        Editable editable = binding.fileName.getText();
   575	        String fileName = (editable == null ? null : editable.toString());
   576	        if (!replace && viewModel.fileExists(fileName)) {
   577	            showReplaceFileDialog();
   578	            return;
   579	        }
   580	
   581	        Intent i = new Intent();
   582	        try {
   583	            i.setData(viewModel.createFile(fileName));
   584	
   585	        } catch (SecurityException e) {
   586	            permissionDeniedToast();
   587	            return;
   588	        }
   589	        setResult(RESULT_OK, i);
   590	        finish();
   591	    }

   595	        try {
   596	            i.setData(viewModel.getFileUri(fileName));
   597	
   598	        } catch (SecurityException e) {
   599	            permissionDeniedToast();
   600	            return;
   601	        }
   602	        setResult(RESULT_OK, i);
   603	        finish();
   604	    }

   606	    private void showReplaceFileDialog() {
   607	        if (getSupportFragmentManager().findFragmentByTag(TAG_REPLACE_FILE_DIALOG) == null) {
   608	            BaseAlertDialog replaceFileDialog = BaseAlertDialog.newInstance(
   609	                    getString(R.string.replace_file),
   610	                    getString(R.string.error_file_exists),
   611	                    0,
   612	                    getString(R.string.yes),
   613	                    getString(R.string.no),
   614	                    null,
   615	                    true);
   616	
   617	            replaceFileDialog.show(getSupportFragmentManager(), TAG_REPLACE_FILE_DIALOG);
   618	        }
   619	    }

   621	    private boolean checkFileNameField() {
   622	        if (TextUtils.isEmpty(binding.fileName.getText())) {
   623	            binding.layoutFileName.setErrorEnabled(true);
   624	            binding.layoutFileName.setError(getString(R.string.file_name_is_empty));
   625	            binding.layoutFileName.requestFocus();
   626	
   627	            return false;
   628	        }
   629	
   630	        binding.layoutFileName.setErrorEnabled(false);
   631	        binding.layoutFileName.setError(null);
   632	
   633	        return true;
   634	    }

   636	    final synchronized void reloadSpinner() {
   637	        if (storageAdapter == null || adapter == null)
   638	            return;
   639	
   640	        List<FileManagerSpinnerAdapter.StorageSpinnerItem> spinnerItems = viewModel.getStorageList();
   641	        spinnerPos = getSpinnerPos(spinnerItems);
   642	        binding.storageSpinner.setSelection(spinnerPos);
   643	
   644	        storageAdapter.clear();
   645	        storageAdapter.addItems(spinnerItems);
   646	        storageAdapter.setTitle(viewModel.curDir.get());
   647	        storageAdapter.notifyDataSetChanged();
   648	
   649	        viewModel.refreshCurDirectory();
   650	    }
   651	
   652	    private int getSpinnerPos(List<FileManagerSpinnerAdapter.StorageSpinnerItem> spinnerItems) {
   653	        for (int i = 0; i < spinnerItems.size(); i++) {
   654	            String curDir = viewModel.curDir.get();
   655	            if (curDir != null && curDir.startsWith(spinnerItems.get(i).getStoragePath())) {
   656	                return i;
   657	            }
   658	        }
   659	
   660	        return -1;
   661	    }

   663	    @Override
   664	    public void onFragmentFinished(@NonNull Fragment f, Intent intent, @NonNull ResultCode code) {
   665	        if (!intent.hasExtra(TAG_URI) || code != ResultCode.OK) {
   666	            return;
   667	        }
   668	
   669	        Uri uri = intent.getParcelableExtra(TAG_URI);
   670	        if (uri.getPath() != null) {
   671	            goToDirectory(uri.getPath());
   672	        }
   673	    }
