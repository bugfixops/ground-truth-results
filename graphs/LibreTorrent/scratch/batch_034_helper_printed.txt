   102	        ++numChangedChildren;

   105	        if (allChildrenChanged)

    46	    public static Pair<TorrentContentFileTree, TorrentContentFileTree[]> buildFileTree(List<BencodeFileItem> files)
    47	    {
    48	        TorrentContentFileTree root = new TorrentContentFileTree(FileTree.ROOT, 0L, FileNode.Type.DIR);
    49	        TorrentContentFileTree parentTree = root;
    50	        TorrentContentFileTree[] leaves = new TorrentContentFileTree[files.size()];
    51	        /* It allows reduce the number of iterations on the paths with equal beginnings */
    52	        String prevPath = "";
    53	        List<BencodeFileItem> filesCopy = new ArrayList<>(files);
    54	        /* Sort reduces the returns number to root */
    55	        Collections.sort(filesCopy);
    56	
    57	        for (BencodeFileItem file : filesCopy) {
    58	            String path;
    59	            /*
    60	             * Compare previous path with new path.
    61	             * Example:
    62	             * prev = dir1/dir2/
    63	             * cur  = dir1/dir2/file1
    64	             *        |________|
    65	             *          equal
    66	             *
    67	             * prev = dir1/dir2/
    68	             * cur  = dir3/file2
    69	             *        |________|
    70	             *         not equal
    71	             */
    72	            if (!prevPath.isEmpty() &&
    73	                    file.getPath().regionMatches(true, 0, prevPath, 0, prevPath.length())) {
    74	                /*
    75	                 * If beginning paths are equal, remove previous path from the new path.
    76	                 * Example:
    77	                 * prev = dir1/dir2/
    78	                 * cur  = dir1/dir2/file1
    79	                 * new  = file1
    80	                 */
    81	                path = file.getPath().substring(prevPath.length());
    82	            } else {
    83	                /* If beginning paths are not equal, return to root */
    84	                path = file.getPath();
    85	                parentTree = root;
    86	            }
    87	
    88	            String[] nodes = path.split(File.separator);
    89	            /*
    90	             * Remove last node (file) from previous path.
    91	             * Example:
    92	             * cur = dir1/dir2/file1
    93	             * new = dir1/dir2/
    94	             */
    95	            prevPath = file.getPath()
    96	                    .substring(0, file.getPath().length() - nodes[nodes.length - 1].length());
    97	
    98	            /* Iterates path nodes */
    99	            for (int i = 0; i < nodes.length; i++) {
   100	                if (!parentTree.contains(nodes[i])) {
   101	                    TorrentContentFileTree leaf = makeObject(file.getIndex(), nodes[i],
   102	                                                            file.getSize(), parentTree,
   103	                                                            i == (nodes.length - 1));
   104	                    leaves[file.getIndex()] = leaf;
   105	                    /* The last leaf item is a file */
   106	                    parentTree.addChild(leaf);
   107	                }
   108	
   109	                TorrentContentFileTree nextParent = parentTree.getChild(nodes[i]);
   110	                /* Skipping leaf nodes */
   111	                if (!nextParent.isFile())
   112	                    parentTree = nextParent;
   113	            }
   114	        }
   115	
   116	        return Pair.create(root, leaves);
   117	    }
