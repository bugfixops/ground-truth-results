    20	package org.proninyaroslav.libretorrent.ui;
    21	
    22	import android.app.Dialog;
    23	import android.os.Build;
    24	import android.os.Bundle;
    25	
    26	import androidx.annotation.NonNull;
    27	
    28	import org.proninyaroslav.libretorrent.R;
    29	
    30	public class PermissionDeniedDialog extends BaseAlertDialog {
    31	    public static PermissionDeniedDialog newInstance() {
    32	        PermissionDeniedDialog frag = new PermissionDeniedDialog();
    33	
    34	        Bundle args = new Bundle();
    35	        frag.setArguments(args);
    36	
    37	        return frag;
    38	    }
    39	
    40	    @NonNull
    41	    @Override
    42	    public Dialog onCreateDialog(Bundle savedInstanceState) {
    43	        super.onCreateDialog(savedInstanceState);
    44	
    45	        String title = getString(R.string.perm_denied_title);
    46	        String message = Build.VERSION.SDK_INT >= Build.VERSION_CODES.R ?
    47	                getString(R.string.perm_denied_warning_android_r) :
    48	                getString(R.string.perm_denied_warning);
    49	        String positiveText = getString(R.string.yes);
    50	        String negativeText = getString(R.string.no);
    51	
    52	        return buildDialog(
    53	                title,
    54	                message,
    55	                null,
    56	                positiveText,
    57	                negativeText,
    58	                null,
    59	                false
    60	        );
    61	    }
    62	}
