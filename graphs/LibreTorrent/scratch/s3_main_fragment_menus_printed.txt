   612	
   613	        disposables.add(Observable.fromIterable(selections)
   614	                .map((selection -> selection.torrentId))

   616	                .subscribe((ids) -> viewModel.forceAnnounceTorrents(ids)));
   617	    }
   618	
   157	        selectionTracker = new SelectionTracker.Builder<>(
   158	                SELECTION_TRACKER_ID,
   159	                binding.torrentList,
   160	                new TorrentListAdapter.KeyProvider(adapter),
   161	                new TorrentListAdapter.ItemLookup(binding.torrentList),
   162	                StorageStrategy.createParcelableStorage(TorrentListItem.class))
   163	                .withSelectionPredicate(SelectionPredicates.createSelectAnything())
   164	                .build();
   165	
   166	        selectionTracker.addObserver(new SelectionTracker.SelectionObserver<TorrentListItem>() {
   167	            @Override
   168	            public void onSelectionChanged()
   169	            {
   170	                super.onSelectionChanged();
   171	
   172	                if (selectionTracker.hasSelection() && actionMode == null) {
   173	                    actionMode = activity.startSupportActionMode(actionModeCallback);
   174	                    setActionModeTitle(selectionTracker.getSelection().size());
   175	
   176	                } else if (!selectionTracker.hasSelection()) {
   177	                    if (actionMode != null)
   178	                        actionMode.finish();
   179	                    actionMode = null;
   180	
   181	                } else {
   182	                    setActionModeTitle(selectionTracker.getSelection().size());
   183	                }
   184	            }
   185	
   186	            @Override
   187	            public void onSelectionRestored()
   188	            {
   189	                super.onSelectionRestored();
   190	
   191	                actionMode = activity.startSupportActionMode(actionModeCallback);
   192	                setActionModeTitle(selectionTracker.getSelection().size());
   193	            }
   194	        });
