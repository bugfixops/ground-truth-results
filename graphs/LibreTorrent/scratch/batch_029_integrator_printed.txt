   122	    private double calcRelevance(AdvancedPeerInfo peer, TorrentStatus torrentStatus)
   123	    {
   124	        double relevance = 0.0;
   125	        PieceIndexBitfield allPieces = torrentStatus.pieces();
   126	        PieceIndexBitfield peerPieces = peer.pieces();
   127	
   128	        int remoteHaves = 0;
   129	        int localMissing = 0;
   130	        for (int i = 0; i < allPieces.size(); i++) {
   131	            if (!allPieces.getBit(i)) {
   132	                ++localMissing;
   133	                if (peerPieces.getBit(i))
   134	                    ++remoteHaves;
   135	            }
   136	        }
   137	        if (localMissing != 0)
   138	            relevance = (double) remoteHaves / (double) localMissing;
   139	
   140	        return relevance;
   141	    }
