   140	    private void subscribeAlertDialog()
   141	    {
   142	        Disposable d = dialogViewModel.observeEvents()
   143	                .subscribe((event) -> {
   144	                    if (event.dialogTag == null)
   145	                        return;
   146	
   147	                    switch (event.type) {
   148	                        case POSITIVE_BUTTON_CLICKED:
   149	                            if (event.dialogTag.equals(TAG_ERROR_REPORT_DIALOG) && errReportDialog != null) {
   150	                                Dialog dialog = errReportDialog.getDialog();
   151	                                if (dialog != null) {
   152	                                    TextInputEditText editText = dialog.findViewById(R.id.comment);
   153	                                    Editable e = editText.getText();
   154	                                    String comment = (e == null ? null : e.toString());
   155	
   156	                                    Utils.reportError(viewModel.errorReport, comment);
   157	                                    errReportDialog.dismiss();
   158	                                }
   159	                            }
   160	                            break;
   161	                        case NEGATIVE_BUTTON_CLICKED:
   162	                            if (event.dialogTag.equals(TAG_ERROR_REPORT_DIALOG) && errReportDialog != null)
   163	                                errReportDialog.dismiss();
   164	                            break;
   165	                    }
   166	                });
   167	        disposables.add(d);
   168	    }

   280	    private void initAlertDialog(View view)
   281	    {
   282	        alert = new AlertDialog.Builder(activity)
   283	                .setTitle(R.string.create_torrent)
   284	                .setNegativeButton(R.string.cancel, null)
   285	                .setPositiveButton(R.string.add, null)
   286	                .setView(view)
   287	                .create();
   288	
   289	
   290	        alert.setCanceledOnTouchOutside(false);
   291	        alert.setOnShowListener((DialogInterface dialog) -> {
   292	            Button cancelButton = alert.getButton(AlertDialog.BUTTON_NEGATIVE);
   293	            Button addButton = alert.getButton(AlertDialog.BUTTON_POSITIVE);
   294	
   295	            viewModel.getState().observe(this, (state) -> {
   296	                if (state.status == CreateTorrentViewModel.BuildState.Status.BUILDING) {
   297	                    alert.setTitle(R.string.creating_torrent_progress);
   298	                    addButton.setVisibility(View.GONE);
   299	                } else {
   300	                    alert.setTitle(R.string.create_torrent);
   301	                    addButton.setVisibility(View.VISIBLE);
   302	                }
   303	
   304	                if (state.status == CreateTorrentViewModel.BuildState.Status.FINISHED)
   305	                    handleFinish();
   306	
   307	                else if (state.status == CreateTorrentViewModel.BuildState.Status.ERROR)
   308	                    handleBuildError(state.err);
   309	            });
   310	
   311	            cancelButton.setOnClickListener((v) ->
   312	                    finish(new Intent(), FragmentCallback.ResultCode.CANCEL));
   313	            addButton.setOnClickListener((v) -> {
   314	                if(TextUtils.isEmpty(viewModel.mutableParams.getSeedPathName()) || "null".equals(viewModel.mutableParams.getSeedPathName())) {
   315	                    binding.layoutFileOrDirPath.setErrorEnabled(true);
   316	                    binding.layoutFileOrDirPath.setError(getText(R.string.error_please_select_file_or_folder_for_seeding_first));
   317	                }
   318	                else{
   319	                    choosePathToSaveDialog();
   320	                }
   321	            });
   322	        });
   323	    }

   348	    private void handleBuildError(Throwable err)
   349	    {
   350	        if (err == null)
   351	            return;
   352	
   353	        Log.e(TAG, Log.getStackTraceString(err));
   354	
   355	        if (err instanceof CreateTorrentViewModel.InvalidTrackerException) {
   356	            binding.layoutTrackerUrls.setErrorEnabled(true);
   357	            binding.layoutTrackerUrls.setError(getString(R.string.invalid_url,
   358	                    ((CreateTorrentViewModel.InvalidTrackerException)err).url));
   359	            binding.layoutTrackerUrls.requestFocus();
   360	
   361	        } else if (err instanceof CreateTorrentViewModel.InvalidWebSeedException) {
   362	            binding.layoutWebSeedUrls.setErrorEnabled(true);
   363	            binding.layoutWebSeedUrls.setError(getString(R.string.invalid_url,
   364	                    ((CreateTorrentViewModel.InvalidWebSeedException)err).url));
   365	            binding.layoutWebSeedUrls.requestFocus();
   366	
   367	        } else if (err instanceof FileNotFoundException) {
   368	            fileOrFolderNotFoundDialog((FileNotFoundException)err);
   369	
   370	        } else if (err instanceof IOException) {
   371	            if (err.getMessage().contains("content total size can't be 0"))
   372	                emptyFolderErrorDialog();
   373	            else
   374	                errorReportDialog(err);
   375	        } else {
   376	            errorReportDialog(err);
   377	        }
   378	    }
   379	
   380	    private void handleFinish()
   381	    {
   382	        Uri savePath = viewModel.mutableParams.getSavePath();
   383	        if (savePath != null) {
   384	            Toast.makeText(activity.getApplicationContext(),
   385	                    getString(R.string.torrent_saved_to, savePath.getPath()),
   386	                    Toast.LENGTH_SHORT)
   387	                    .show();
   388	        }
   389	        if (viewModel.mutableParams.isStartSeeding()) {
   390	            try {
   391	                disposables.add(viewModel.downloadTorrent()
   392	                        .subscribeOn(Schedulers.io())
   393	                        .subscribe(
   394	                                () -> finish(new Intent(), FragmentCallback.ResultCode.OK),
   395	                                this::handleBuildError)
   396	                );
   397	            } catch (UnknownUriException e) {
   398	                handleBuildError(e);
   399	            }
   400	        } else {
   401	            finish(new Intent(), FragmentCallback.ResultCode.OK);
   402	        }
   403	    }

   405	    private void openPathErrorDialog(boolean isFile)
   406	    {
   407	        if (!isAdded())
   408	            return;
   409	
   410	        FragmentManager fm = getChildFragmentManager();
   411	        if (fm.findFragmentByTag(TAG_ERROR_FOLDER_IS_EMPTY) == null) {
   412	            BaseAlertDialog errDialog = BaseAlertDialog.newInstance(
   413	                    getString(R.string.error),
   414	                    getString(isFile ? R.string.unable_to_open_file : R.string.unable_to_open_folder),
   415	                    0,
   416	                    getString(R.string.ok),
   417	                    null,
   418	                    null,
   419	                    true);
   420	
   421	            errDialog.show(fm, TAG_ERROR_FOLDER_IS_EMPTY);
   422	        }
   423	    }

   476	            errReportDialog.show(fm, TAG_ERROR_REPORT_DIALOG);
   477	        }
   478	    }
