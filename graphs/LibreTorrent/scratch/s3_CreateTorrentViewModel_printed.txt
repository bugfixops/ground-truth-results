    67	    public static class BuildState

    78	        }

    91	    public static class InvalidWebSeedException extends Exception

   101	    public CreateTorrentViewModel(@NonNull Application application)
   102	    {
   103	        super(application);
   104	
   105	        engine = TorrentEngine.getInstance(application);
   106	        disposables.add(engine.observeNeedStartEngine()
   107	                .subscribeOn(Schedulers.io())
   108	                .filter((needStart) -> needStart)
   109	                .subscribe((needStart) -> engine.start()));
   110	
   111	        fs = SystemFacadeHelper.getFileSystemFacade(application);
   112	        mutableParams.getSeedPath().addOnPropertyChangedCallback(dirPathCallback);
   113	        state.setValue(new BuildState(BuildState.Status.UNKNOWN, null));
   114	        buildProgress.setValue(0);
   115	    }

   117	    @Override
   118	    protected void onCleared()
   119	    {
   120	        super.onCleared();
   121	
   122	        mutableParams.getSeedPath().removeOnPropertyChangedCallback(dirPathCallback);
   123	        disposables.clear();
   124	    }

   136	    public void setPiecesSizeIndex(int index)
   137	    {
   138	        if (index < 0 || index >= engine.getPieceSizeList().length)
   139	            return;

   141	        mutableParams.setPieceSizeIndex(index);

   145	        if (index < 0 || index >= engine.getTorrentVersionList().length)
   146	            return;

   148	        mutableParams.setTorrentVersionIndex(index);
   149	    }

   151	    public void buildTorrent()
   152	    {
   153	        state.setValue(new BuildState(BuildState.Status.BUILDING, null));
   154	
   155	        TorrentBuilder builder;
   156	        try {
   157	            builder = makeBuilder();
   158	
   159	        } catch (Exception e) {
   160	            onBuildError(e);
   161	
   162	            return;
   163	        }
   164	
   165	        resetPercentProgress();
   166	        disposables.add(builder.observeProgress()
   167	                .subscribeOn(Schedulers.io())
   168	                .filter(Objects::nonNull)
   169	                .subscribe(this::makePercentProgress)
   170	        );
   171	        disposables.add(builder.build()
   172	                .subscribeOn(Schedulers.io())
   173	                .subscribe(this::onBuildSuccess, this::onBuildError)
   174	        );
   175	    }
   176	
   177	    private TorrentBuilder makeBuilder() throws Exception
   178	    {
   179	        Uri seedPath = mutableParams.getSeedPath().get();
   180	        if (seedPath == null)
   181	            throw new IllegalArgumentException("Seed path is null");
   182	        Uri savePath = mutableParams.getSavePath();
   183	        if (savePath == null)
   184	            throw new IllegalArgumentException("Save path is null");
   185	
   186	        if (!Utils.isFileSystemPath(seedPath))
   187	            throw new IllegalArgumentException("SAF doesn't supported");
   188	
   189	        return new TorrentBuilder(getApplication())
   190	                .setSeedPath(seedPath)
   191	                .setPieceSize(getPieceSizeByIndex(mutableParams.getPieceSizeIndex()))
   192	                .setTorrentVersion(getTorrentVersionByIndex(mutableParams.getTorrentVersionIndex()))
   193	                .addTrackers(getAndValidateTrackers())
   194	                .addUrlSeeds(getAndValidateWebSeeds())
   195	                .setAsPrivate(mutableParams.isPrivateTorrent())
   196	                .setCreator(makeCreator())
   197	                .setComment(mutableParams.getComments())
   198	                .setFileNameFilter((fileName) -> {
   199	                    List<String> skipFilesList = decodeSkipFilesList();
   200	                    if (skipFilesList.isEmpty())
   201	                        return true;
   202	
   203	                    for (String skipFile : skipFilesList)
   204	                        if (fileName.toLowerCase().endsWith(skipFile.toLowerCase().trim()))
   205	                            return false;
   206	
   207	                    return true;
   208	                });
   209	    }
   210	
   211	    private void onBuildSuccess(byte[] bencode)
   212	    {
   213	        Uri savePath = mutableParams.getSavePath();
   214	        if (savePath != null) {
   215	            try {
   216	                fs.write(bencode, savePath);
   217	
   218	            } catch (IOException | UnknownUriException e) {
   219	                onBuildError(e);
   220	
   221	                return;
   222	            }
   223	        }
   224	
   225	        state.postValue(new BuildState(BuildState.Status.FINISHED, null));
   226	    }
   227	
   228	    private void onBuildError(Throwable e)
   229	    {
   230	        Uri savePath = mutableParams.getSavePath();
   231	        if (savePath != null) {
   232	            try {
   233	                fs.deleteFile(savePath);
   234	            } catch (IOException | UnknownUriException eio) {
   235	                /* Ignore */
   236	            }
   237	        }
   238	
   239	        state.postValue(new BuildState(BuildState.Status.ERROR, e));
   240	    }

   242	    private void makePercentProgress(TorrentBuilder.Progress progress)
   243	    {
   244	        buildProgress.postValue((int)(progress.piece * 100.0) / progress.numPieces);
   245	    }
   246	
   247	    private void resetPercentProgress()
   248	    {
   249	        buildProgress.postValue(0);
   250	    }

   252	    private String makeCreator()
   253	    {
   254	        Context context = getApplication();
   255	        String creator = context.getString(R.string.app_name);
   256	        String versionName = SystemFacadeHelper.getSystemFacade(context)
   257	                .getAppVersionName();
   258	        if (versionName == null)
   259	            return creator;
   260	
   261	        return creator + " " + versionName;
   262	    }

   264	    private List<TorrentBuilder.Tracker> getAndValidateTrackers() throws InvalidTrackerException
   265	    {
   266	        List<TorrentBuilder.Tracker> validTrackers = new ArrayList<>();
   267	        int tier = 0;
   268	        for (String url : decodeUrls(mutableParams.getTrackerUrls())) {
   269	            try {
   270	                url = normalizeAndValidateUrl(url);
   271	
   272	            } catch (IllegalArgumentException e) {
   273	                throw new InvalidTrackerException(url);
   274	            }
   275	            validTrackers.add(new TorrentBuilder.Tracker(url, tier++));
   276	        }
   277	
   278	        return validTrackers;
   279	    }
   280	
   281	    private List<String> getAndValidateWebSeeds() throws InvalidWebSeedException
   282	    {
   283	        List<String> validWebSeeds = new ArrayList<>();
   284	        for (String url : decodeUrls(mutableParams.getWebSeedUrls())) {
   285	            try {
   286	                url = normalizeAndValidateUrl(url);
   287	
   288	            } catch (IllegalArgumentException e) {
   289	                throw new InvalidWebSeedException(url);
   290	            }
   291	            validWebSeeds.add(url);
   292	        }
   293	
   294	        return validWebSeeds;
   295	    }
   296	
   297	    private String[] decodeUrls(String urlsStr)
   298	    {
   299	        String[] urls = new String[0];
   300	        if (!TextUtils.isEmpty(urlsStr))
   301	            urls = urlsStr.split("\n");
   302	
   303	        return urls;
   304	    }
   305	
   306	    private String normalizeAndValidateUrl(String url)
   307	    {
   308	        NormalizeUrl.Options options = new NormalizeUrl.Options();
   309	        options.decode = false;
   310	        try {
   311	            url = NormalizeUrl.normalize(url, options);
   312	
   313	        } catch (NormalizeUrlException e) {
   314	            throw new IllegalArgumentException();
   315	        }
   316	
   317	        if (!Utils.isValidTrackerUrl(url))
   318	            throw new IllegalArgumentException();
   319	
   320	        return url;
   321	    }
   322	
   323	    private List<String> decodeSkipFilesList()
   324	    {
   325	        List<String> skipFilesList = new ArrayList<>();
   326	        if (!TextUtils.isEmpty(mutableParams.getSkipFiles()))
   327	            skipFilesList = new ArrayList<>(Arrays.asList(mutableParams.getSkipFiles()
   328	                    .split(CreateTorrentMutableParams.FILTER_SEPARATOR)));
   329	
   330	        return skipFilesList;
   331	    }
   332	
   333	    private int getPieceSizeByIndex(int index)
   334	    {
   335	        return engine.getPieceSizeList()[index] * 1024;
   336	    }
   337	
   338	    private TorrentBuilder.TorrentVersion getTorrentVersionByIndex(int index) {
   339	        int version = engine.getTorrentVersionList()[index];
   340	        if (version == 0) {
   341	            return TorrentBuilder.TorrentVersion.HYBRID;
   342	        } else if (version == 1) {
   343	            return TorrentBuilder.TorrentVersion.V1_ONLY;
   344	        } else if (version == 2) {
   345	            return TorrentBuilder.TorrentVersion.V2_ONLY;
   346	        } else {
   347	            Log.e(TAG, "Unknown torrent version " + version);
   348	            return TorrentBuilder.TorrentVersion.HYBRID;
   349	        }
   350	    }

   352	    public Completable downloadTorrent() throws UnknownUriException {
   353	        /* Use seed path parent; otherwise use save torrent file path */
   354	        Uri savePath;
   355	        Uri seedPath = mutableParams.getSeedPath().get();
   356	        if (seedPath != null) {
   357	            savePath = fs.getParentDirUri(seedPath);
   358	            if (savePath == null)
   359	                savePath = mutableParams.getSavePath();
   360	        } else {
   361	            savePath = mutableParams.getSavePath();
   362	        }
   363	        Uri torrentFilePath = mutableParams.getSavePath();
   364	        if (savePath == null || torrentFilePath == null)
   365	            return Completable.complete();
   366	
   367	        Uri path = savePath;
   368	        return Completable.create((emitter) -> {
   369	            if (emitter.isDisposed())
   370	                return;
   371	
   372	            Disposable d = engine.observeEngineRunning()
   373	                    .subscribeOn(Schedulers.io())
   374	                    .subscribe((isRunning) -> {
   375	                        if (isRunning) {
   376	                            engine.addTorrent(torrentFilePath, path);
   377	                            if (!emitter.isDisposed())
   378	                                emitter.onComplete();
   379	                        }
   380	                    });
   381	
   382	            emitter.setDisposable(d);
   383	        });
   384	    }

   386	    public void finish()
   387	    {
   388	        disposables.clear();
   389	    }

   391	    private Observable.OnPropertyChangedCallback dirPathCallback = new Observable.OnPropertyChangedCallback()
   392	    {
   393	        @Override
   394	        public void onPropertyChanged(Observable sender, int propertyId)
   395	        {
   396	            Uri seedPath = mutableParams.getSeedPath().get();
   397	            if (seedPath == null)
   398	                return;
   399	            try {
   400	                mutableParams.setSeedPathName(fs.getDirPath(seedPath));
   401	            } catch (UnknownUriException e) {
   402	                Log.e(TAG, Log.getStackTraceString(e));
   403	            }
   404	        }
   405	    };
