   138	    private void subscribeAlertDialog()
   139	    {
   140	        Disposable d = dialogViewModel.observeEvents()
   141	                .subscribe((event) -> {
   142	                    if (event.dialogTag == null)
   143	                        return;
   144	
   145	                    switch (event.type) {
   146	                        case POSITIVE_BUTTON_CLICKED:
   147	                            if (event.dialogTag.equals(TAG_BACKUP_FEEDS_ERROR_REPORT_DIALOG) &&
   148	                                    backupFeedsReportDialog != null) {
   149	                                sendErrorReport(backupFeedsReportDialog.getDialog());
   150	                                backupFeedsReportDialog.dismiss();
   151	
   152	                            } else if (event.dialogTag.equals(TAG_RESTORE_FEEDS_ERROR_REPORT_DIALOG) &&
   153	                                    restoreFeedsReportDialog!= null) {
   154	                                sendErrorReport(restoreFeedsReportDialog.getDialog());
   155	                                restoreFeedsReportDialog.dismiss();
   156	                            }
   157	                            break;
   158	                        case NEGATIVE_BUTTON_CLICKED:
   159	                            if (event.dialogTag.equals(TAG_BACKUP_FEEDS_ERROR_REPORT_DIALOG) &&
   160	                                    backupFeedsReportDialog!= null) {
   161	                                backupFeedsReportDialog.dismiss();
   162	
   163	                            } else if (event.dialogTag.equals(TAG_RESTORE_FEEDS_ERROR_REPORT_DIALOG) &&
   164	                                    restoreFeedsReportDialog!= null) {
   165	                                restoreFeedsReportDialog.dismiss();
   166	                            }
   167	                    }
   168	                });
   169	        disposables.add(d);
   170	    }
   171	
   172	    private void subscribeMsgViewModel()
   173	    {
   174	        disposables.add(msgViewModel.observeFeedItemsOpened()
   175	                .observeOn(AndroidSchedulers.mainThread())
   176	                .subscribe(this::showFeedItems));
   177	
   178	        disposables.add(msgViewModel.observeFeedItemsClosed()
   179	                .observeOn(AndroidSchedulers.mainThread())
   180	                .subscribe((__) -> showBlankFragment()));
   181	
   182	        disposables.add(msgViewModel.observeFeedsDeleted()
   183	                .observeOn(AndroidSchedulers.mainThread())
   184	                .subscribe((ids) -> {
   185	                    FeedItemsFragment f = getCurrentFeedItemsFragment();
   186	                    if (f != null && ids.contains(f.getFeedId()))
   187	                        showBlankFragment();
   188	                }));
   189	    }
   190	
   191	    private void showBlankFragment()
   192	    {
   193	        if (Utils.isTwoPane(this)) {
   194	            FragmentManager fm = getSupportFragmentManager();
   195	            BlankFragment blank = BlankFragment.newInstance(getString(R.string.select_or_add_feed_channel));
   196	            fm.beginTransaction()
   197	                    .replace(R.id.feed_items_fragmentContainer, blank)
   198	                    .setTransition(FragmentTransaction.TRANSIT_FRAGMENT_CLOSE)
   199	                    .commitAllowingStateLoss();
   200	        }
   201	    }
   202	
   203	    private void showFeedItems(long feedId)
   204	    {
   205	        if (Utils.isTwoPane(this)) {
   206	            FragmentManager fm = getSupportFragmentManager();
   207	            FeedItemsFragment items = FeedItemsFragment.newInstance(feedId);
   208	            Fragment fragment = fm.findFragmentById(R.id.feed_items_fragmentContainer);
   209	
   210	            if (fragment instanceof FeedItemsFragment) {
   211	                long oldId = ((FeedItemsFragment)fragment).getFeedId();
   212	                if (feedId == oldId)
   213	                    return;
   214	            }
   215	            fm.beginTransaction()
   216	                    .replace(R.id.feed_items_fragmentContainer, items)
   217	                    .setTransition(FragmentTransaction.TRANSIT_FRAGMENT_FADE)
   218	                    .commit();
   219	
   220	        } else {
   221	            Intent i = new Intent(this, FeedItemsActivity.class);
   222	            i.putExtra(FeedItemsActivity.TAG_FEED_ID, feedId);
   223	            startActivity(i);
   224	        }
   225	    }
   226	
   227	    private FeedItemsFragment getCurrentFeedItemsFragment()
   228	    {
   229	        if (!Utils.isTwoPane(this))
   230	            return null;
   231	
   232	        FragmentManager fm = getSupportFragmentManager();
   233	        Fragment fragment = fm.findFragmentById(R.id.feed_items_fragmentContainer);
   234	
   235	        return (fragment instanceof FeedItemsFragment ? (FeedItemsFragment)fragment : null);
   236	    }

   247	    @Override
   248	    public boolean onOptionsItemSelected(MenuItem item)
   249	    {
   250	        int itemId = item.getItemId();
   251	        if (itemId == R.id.refresh_feed_channel_menu) {
   252	            viewModel.refreshAllFeeds();
   253	        } else if (itemId == R.id.backup_feed_channels_menu) {
   254	            backupFeedsChooseDialog();
   255	        } else if (itemId == R.id.restore_feed_channels_backup_menu) {
   256	            restoreFeedsChooseDialog();
   257	        }
   258	
   259	        return true;
   260	    }

   262	    private void backupFeedsChooseDialog()
   263	    {
   264	        Intent i = new Intent(this, FileManagerDialog.class);
   265	        FileSystemFacade fs = SystemFacadeHelper.getFileSystemFacade(getApplicationContext());
   266	        FileManagerConfig config = new FileManagerConfig(fs.getUserDirPath(),
   267	                null,
   268	                FileManagerConfig.SAVE_FILE_MODE);
   269	        config.fileName = "Feeds-" + new SimpleDateFormat("yyyy_MM_dd_HH_mm_ss", Locale.getDefault())
   270	                .format(new Date(System.currentTimeMillis())) +
   271	                "." + FeedRepositoryImpl.SERIALIZE_FILE_FORMAT;
   272	        config.mimeType = FeedRepositoryImpl.SERIALIZE_MIME_TYPE;
   273	
   274	        i.putExtra(FileManagerDialog.TAG_CONFIG, config);
   275	        backupFeedsChoose.launch(i);
   276	    }
   277	
   278	    private void restoreFeedsChooseDialog()
   279	    {
   280	        Intent i = new Intent(this, FileManagerDialog.class);
   281	        FileSystemFacade fs = SystemFacadeHelper.getFileSystemFacade(getApplicationContext());
   282	        FileManagerConfig config = new FileManagerConfig(fs.getUserDirPath(),
   283	                getString(R.string.feeds_backup_selection_dialog_title),
   284	                FileManagerConfig.FILE_CHOOSER_MODE);
   285	        config.highlightFileTypes = new ArrayList<>();
   286	        config.highlightFileTypes.add(FeedRepositoryImpl.SERIALIZE_FILE_FORMAT);
   287	
   288	        i.putExtra(FileManagerDialog.TAG_CONFIG, config);
   289	        restoreFeedsBackupChoose.launch(i);
   290	    }

   292	    private void backupFeedsErrorDialog(Throwable e)
   293	    {
   294	        viewModel.errorReport = e;
   295	        FragmentManager fm = getSupportFragmentManager();
   296	        if (fm.findFragmentByTag(TAG_BACKUP_FEEDS_ERROR_REPORT_DIALOG) == null) {
   297	            backupFeedsReportDialog = ErrorReportDialog.newInstance(
   298	                    getString(R.string.error),
   299	                    getString(R.string.error_backup_feeds),
   300	                    (e != null ? Log.getStackTraceString(e) : null));
   301	
   302	            backupFeedsReportDialog.show(fm, TAG_BACKUP_FEEDS_ERROR_REPORT_DIALOG);
   303	        }
   304	    }
   305	
   306	    private void restoreFeedsBackupErrorDialog(Throwable e)
   307	    {
   308	        FragmentManager fm = getSupportFragmentManager();
   309	        if (e instanceof JsonSyntaxException) {
   310	            if (fm.findFragmentByTag(TAG_RESTORE_FEEDS_ERROR_DIALOG) == null) {
   311	                BaseAlertDialog errDialog = BaseAlertDialog.newInstance(
   312	                        getString(R.string.error),
   313	                        getString(R.string.error_import_invalid_format),
   314	                        0,
   315	                        getString(R.string.ok),
   316	                        null,
   317	                        null,
   318	                        true);
   319	                errDialog.show(fm, TAG_RESTORE_FEEDS_ERROR_DIALOG);
   320	            }
   321	        } else {
   322	            viewModel.errorReport = e;
   323	            if (fm.findFragmentByTag(TAG_RESTORE_FEEDS_ERROR_REPORT_DIALOG) == null) {
   324	                restoreFeedsReportDialog = ErrorReportDialog.newInstance(
   325	                        getString(R.string.error),
   326	                        getString(R.string.error_restore_feeds_backup),
   327	                        (e != null ? Log.getStackTraceString(e) : null));
   328	
   329	                restoreFeedsReportDialog.show(fm, TAG_RESTORE_FEEDS_ERROR_REPORT_DIALOG);
   330	            }
   331	        }
   332	    }
   333	
   334	    private void sendErrorReport(Dialog errDialog)
   335	    {
   336	        if (errDialog == null)
   337	            return;
   338	
   339	        TextInputEditText editText = errDialog.findViewById(R.id.comment);
   340	        Editable e = editText.getText();
   341	        String comment = (e == null ? null : e.toString());
   342	
   343	        Utils.reportError(viewModel.errorReport, comment);
   344	    }

   346	    private void backupFeeds(Uri file)
   347	    {
   348	        disposables.add(Completable.fromCallable(() -> {
   349	            viewModel.saveFeedsSync(file);
   350	            return true;
   351	        }).subscribeOn(Schedulers.io())
   352	          .observeOn(AndroidSchedulers.mainThread())
   353	          .subscribe(() -> Toast.makeText(this,
   354	                  R.string.backup_feeds_successfully,
   355	                  Toast.LENGTH_SHORT)
   356	                  .show(), this::restoreFeedsBackupErrorDialog));
   357	    }
   358	
   359	    private void restoreFeedsBackup(Uri file)
   360	    {
   361	        disposables.add(Observable.fromCallable(() -> viewModel.restoreFeedsSync(file))
   362	                .subscribeOn(Schedulers.io())
   363	                .observeOn(AndroidSchedulers.mainThread())
   364	                .subscribe((feedIdList) -> {
   365	                        Toast.makeText(this,
   366	                                R.string.restore_feeds_backup_successfully,
   367	                                Toast.LENGTH_SHORT)
   368	                                .show();
   369	                        viewModel.refreshFeeds(feedIdList);
   370	                    }, this::backupFeedsErrorDialog));
   371	    }

   373	    final ActivityResultLauncher<Intent> backupFeedsChoose = registerForActivityResult(
   374	            new ActivityResultContracts.StartActivityForResult(),
   375	            result -> {
   376	                Intent data = result.getData();
   377	                if (result.getResultCode() != Activity.RESULT_OK)
   378	                    return;
   379	
   380	                if (data == null || data.getData() == null) {
   381	                    backupFeedsErrorDialog(null);
   382	                    return;
   383	                }
   384	
   385	                backupFeeds(data.getData());
   386	            }
   387	    );
   388	
   389	    final ActivityResultLauncher<Intent> restoreFeedsBackupChoose = registerForActivityResult(
   390	            new ActivityResultContracts.StartActivityForResult(),
   391	            result -> {
   392	                Intent data = result.getData();
   393	                if (result.getResultCode() != Activity.RESULT_OK)
   394	                    return;
   395	
   396	                if (data == null || data.getData() == null) {
   397	                    restoreFeedsBackupErrorDialog(null);
   398	                    return;
   399	                }
   400	
   401	                restoreFeedsBackup(data.getData());
   402	            }
   403	    );
