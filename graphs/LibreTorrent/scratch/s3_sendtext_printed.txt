    20	package org.proninyaroslav.libretorrent.ui;
    21	
    22	import android.app.Activity;
    23	import android.content.ClipData;
    24	import android.content.ClipboardManager;
    25	import android.content.Intent;
    26	import android.os.Bundle;
    27	import android.widget.Toast;
    28	
    29	import org.proninyaroslav.libretorrent.R;
    30	
    31	/*
    32	 * Adds "Copy" item in share dialog.
    33	 */
    34	
    35	public class SendTextToClipboard extends Activity
    36	{
    37	    @Override
    38	    public void onCreate(Bundle savedInstanceState)
    39	    {
    40	        super.onCreate(savedInstanceState);
    41	
    42	        Intent intent = getIntent();
    43	
    44	        if (intent.hasExtra(Intent.EXTRA_TEXT)) {
    45	            ClipboardManager clipboard = (ClipboardManager) getSystemService(Activity.CLIPBOARD_SERVICE);
    46	            ClipData clip;
    47	
    48	            CharSequence text = intent.getCharSequenceExtra(Intent.EXTRA_TEXT);
    49	
    50	            clip = ClipData.newPlainText(intent.getType(), text);
    51	            clipboard.setPrimaryClip(clip);
    52	
    53	            Toast.makeText(getApplicationContext(),
    54	                    R.string.text_copied_to_clipboard,
    55	                    Toast.LENGTH_SHORT)
    56	                    .show();
    57	        }
    58	
    59	        finish();
    60	        overridePendingTransition(0, 0);
    61	    }
    62	}
