   490	    private void setInitSelection(List<AbstractTagItem> items) {
   491	        String selectedTagIdStr = PreferenceManager
   492	                .getDefaultSharedPreferences(this)
   493	                .getString(
   494	                        getString(R.string.drawer_tags_selected_item),
   495	                        getString(R.string.tag_empty_item)
   496	                );
   497	        if (selectedTagIdStr.equals(getString(R.string.tag_empty_item))) {
   498	            tagsAdapter.setSelectedItem(new EmptyTagItem());
   499	            viewModel.setTagFilter(TorrentFilterCollection.all(), true);
   500	        } else if (selectedTagIdStr.equals(getString(R.string.tag_no_tags_item))) {
   501	            tagsAdapter.setSelectedItem(new NoTagsItem());
   502	            viewModel.setTagFilter(TorrentFilterCollection.noTags(), true);
   503	        } else {
   504	            long selectedTagId;
   505	            try {
   506	                selectedTagId = Long.parseLong(selectedTagIdStr);
   507	            } catch (NumberFormatException e) {
   508	                Log.e(TAG, "Unable to parse tag id: " + Log.getStackTraceString(e));
   509	                tagsAdapter.setSelectedItem(new EmptyTagItem());
   510	                viewModel.setTagFilter(TorrentFilterCollection.all(), true);
   511	                return;
   512	            }
   513	            disposables.add(Observable.fromIterable(items)
   514	                    .subscribeOn(Schedulers.computation())
   515	                    .filter((item) -> item instanceof TagItem &&
   516	                            ((TagItem) item).info.id == selectedTagId)
   517	                    .observeOn(AndroidSchedulers.mainThread())
   518	                    .subscribe((item) -> {
   519	                        tagsAdapter.setSelectedItem(item);
   520	                        viewModel.setTagFilter(
   521	                                TorrentFilterCollection.tag(((TagItem) item).info),
   522	                                true
   523	                        );
   524	                    })
   525	            );
   526	        }
   527	    }

   295	    private final TagsAdapter.OnClickListener tagsClickListener = new TagsAdapter.OnClickListener() {
   296	        @Override
   297	        public void onTagSelected(@NonNull AbstractTagItem item) {
   298	            if (item instanceof TagItem) {
   299	                viewModel.setTagFilter(
   300	                        TorrentFilterCollection.tag(((TagItem) item).info),
   301	                        true
   302	                );
   303	            } else if (item instanceof EmptyTagItem) {
   304	                viewModel.setTagFilter(TorrentFilterCollection.all(), true);
   305	            } else if (item instanceof NoTagsItem) {
   306	                viewModel.setTagFilter(TorrentFilterCollection.noTags(), true);
   307	            }
   308	
   309	            saveSelectedTag(item);
   310	
   311	            if (drawerLayout != null)
   312	                drawerLayout.closeDrawer(GravityCompat.START);
   313	        }

   315	        @Override
   316	        public void onTagMenuClicked(@NonNull AbstractTagItem abstractItem, int menuId) {
   317	            if (!(abstractItem instanceof TagItem)) {
   318	                return;
   319	            }
   320	            TagItem item = ((TagItem) abstractItem);
   321	            if (menuId == R.id.edit_tag_menu) {
   322	                Intent i = new Intent(MainActivity.this, AddTagActivity.class);
   323	                i.putExtra(AddTagActivity.TAG_INIT_INFO, item.info);
   324	                startActivity(i);
   325	            } else if (menuId == R.id.delete_tag_menu) {
   326	                disposables.add(viewModel.deleteTag(item.info)
   327	                        .subscribeOn(Schedulers.io())
   328	                        .observeOn(AndroidSchedulers.mainThread())
   329	                        .subscribe(
   330	                                () -> {
   331	                                    if (item.isSame(tagsAdapter.getSelectedItem())) {
   332	                                        EmptyTagItem emptyItem = new EmptyTagItem();
   333	                                        saveSelectedTag(emptyItem);
   334	                                        tagsAdapter.setSelectedItem(emptyItem);
   335	                                        viewModel.setTagFilter(TorrentFilterCollection.all(), true);
   336	                                    }
   337	                                },
   338	                                (e) -> {
   339	                                    Log.e(TAG, Log.getStackTraceString(e));
   340	                                    Snackbar.make(
   341	                                            mainFragment.getCoordinatorLayout(),
   342	                                            R.string.tag_deleting_failed,
   343	                                            Snackbar.LENGTH_LONG
   344	                                    ).show();
   345	                                }
   346	                        )
   347	                );
   348	            }
   349	        }
   350	    };
