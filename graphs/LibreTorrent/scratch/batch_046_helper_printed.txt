   194	    @Override
   195	    public Flowable<String> observeSettingsChanged() {
   196	        return Flowable.create((emitter) -> {
   197	            SharedPreferences.OnSharedPreferenceChangeListener listener = (sharedPreferences, key) -> {
   198	                if (!emitter.isCancelled())
   199	                    emitter.onNext(key);
   200	            };
   201	
   202	            if (!emitter.isCancelled()) {
   203	                pref.registerOnSharedPreferenceChangeListener(listener);
   204	                emitter.setDisposable(Disposables.fromAction(() ->
   205	                        pref.unregisterOnSharedPreferenceChangeListener(listener)));
   206	            }
   207	
   208	        }, BackpressureStrategy.LATEST);
   209	    }
