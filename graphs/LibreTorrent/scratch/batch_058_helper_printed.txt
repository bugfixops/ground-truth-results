   164	    public static boolean isNetworkTypeAllowed(@NonNull Context context) {
   165	        SystemFacade systemFacade = SystemFacadeHelper.getSystemFacade(context);
   166	
   167	        SettingsRepository pref = RepositoryHelper.getSettingsRepository(context);
   168	        boolean enableRoaming = pref.enableRoaming();
   169	        boolean unmeteredOnly = pref.unmeteredConnectionsOnly();
   170	
   171	        boolean noUnmeteredOnly;
   172	        boolean noRoaming;
   173	
   174	        NetworkCapabilities caps = systemFacade.getNetworkCapabilities();
   175	        /*
   176	         * Use ConnectivityManager#isActiveNetworkMetered() instead of NetworkCapabilities#NET_CAPABILITY_NOT_METERED,
   177	         * since Android detection VPN as metered, including on Android 9, oddly enough.
   178	         * I think this is due to what VPN services doesn't use setUnderlyingNetworks() method.
   179	         *
   180	         * See for details: https://developer.android.com/about/versions/pie/android-9.0-changes-all#network-capabilities-vpn
   181	         */
   182	        boolean unmetered = caps != null && caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_NOT_METERED) ||
   183	                !systemFacade.isActiveNetworkMetered();
   184	        noUnmeteredOnly = !unmeteredOnly || unmetered;
   185	
   186	        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
   187	            noRoaming = !enableRoaming || caps != null && caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_NOT_ROAMING);
   188	        } else {
   189	            NetworkInfo netInfo = systemFacade.getActiveNetworkInfo();
   190	            noRoaming = netInfo != null && !(enableRoaming && netInfo.isRoaming());
   191	        }
   192	
   193	        return noUnmeteredOnly && noRoaming;
   194	    }

   338	    public static float getBatteryLevel(@NonNull Context context) {
   339	        Intent batteryIntent = context.registerReceiver(null, new IntentFilter(Intent.ACTION_BATTERY_CHANGED));
   340	        if (batteryIntent == null)
   341	            return 50.0f;
   342	        int level = batteryIntent.getIntExtra(BatteryManager.EXTRA_LEVEL, -1);
   343	        int scale = batteryIntent.getIntExtra(BatteryManager.EXTRA_SCALE, -1);
   344	
   345	        /* Error checking that probably isn't needed but I added just in case */
   346	        if (level == -1 || scale == -1)
   347	            return 50.0f;
   348	
   349	        return ((float) level / (float) scale) * 100.0f;

   462	    public static byte[] fetchHttpUrl(@NonNull Context context,
   463	                                      @NonNull String url) throws FetchLinkException {
   464	        byte[][] response = new byte[1][];
   465	
   466	        if (!Utils.checkConnectivity(context))
   467	            throw new FetchLinkException("No network connection");
   468	
   469	        final ArrayList<Throwable> errorArray = new ArrayList<>(1);
   470	        HttpConnection connection;
   471	        try {
   472	            connection = new HttpConnection(url);
   473	        } catch (Exception e) {
   474	            throw new FetchLinkException(e);
   475	        }
   476	
   477	        connection.setListener(new HttpConnection.Listener() {
   478	            @Override
   479	            public void onConnectionCreated(HttpURLConnection conn) {
   480	                /* Nothing */
   481	            }
   482	
   483	            @Override
   484	            public void onResponseHandle(HttpURLConnection conn, int code, String message) {
   485	                if (code == HttpURLConnection.HTTP_OK) {
   486	                    try(InputStream is = conn.getInputStream()) {
   487	                        response[0] = IOUtils.toByteArray(is);
   488	
   489	                    } catch (IOException e) {
   490	                        errorArray.add(e);
   491	                    }
   492	                } else {
   493	                    errorArray.add(new FetchLinkException("Failed to fetch link, response code: " + code));
   494	                }
   495	            }
   496	
   497	            @Override
   498	            public void onMovedPermanently(String newUrl) {
   499	                /* Nothing */
   500	            }
   501	
   502	            @Override
   503	            public void onIOException(IOException e) {
   504	                errorArray.add(e);
   505	            }
   506	
   507	            @Override
   508	            public void onTooManyRedirects() {
   509	                errorArray.add(new FetchLinkException("Too many redirects"));
   510	            }
   511	        });
   512	        connection.run();
   513	
   514	        if (!errorArray.isEmpty()) {
   515	            StringBuilder s = new StringBuilder();
   516	            for (Throwable e : errorArray) {
   517	                String msg = e.getMessage();
   518	                if (msg != null)
   519	                    s.append(msg.concat("\n"));
   520	            }
   521	
   522	            throw new FetchLinkException(s.toString());
   523	        }
   524	
   525	        return response[0];
   526	    }

   544	    public static int[] getVersionComponents(@NonNull String versionName) {
   545	        int[] version = new int[3];
   546	
   547	        /* Discard additional information */
   548	        versionName = getAppVersionNumber(versionName);
   549	
   550	        String[] components = versionName.split("\\.");
   551	        if (components.length < 2)
   552	            return version;
   553	
   554	        try {
   555	            version[0] = Integer.parseInt(components[0]);
   556	            version[1] = Integer.parseInt(components[1]);
   557	            if (components.length >= 3)
   558	                version[2] = Integer.parseInt(components[2]);
   559	
   560	        } catch (NumberFormatException e) {
   561	            /* Ignore */
   562	        }
   563	
   564	        return version;
   565	    }

   567	    public static String makeSha1Hash(@NonNull String s) {
   568	        MessageDigest messageDigest;
   569	        try {
   570	            messageDigest = MessageDigest.getInstance("SHA1");
   571	        } catch (NoSuchAlgorithmException e) {
   572	            return null;
   573	        }
   574	        messageDigest.update(s.getBytes(StandardCharsets.UTF_8));
   575	        StringBuilder sha1 = new StringBuilder();
   576	        for (byte b : messageDigest.digest()) {
   577	            if ((0xff & b) < 0x10)
   578	                sha1.append("0");
   579	            sha1.append(Integer.toHexString(0xff & b));
   580	        }
   581	
   582	        return sha1.toString();
   583	    }

   585	    public static SSLContext getSSLContext() throws GeneralSecurityException {
   586	        TrustManagerFactory tmf = TrustManagerFactory.getInstance(TrustManagerFactory.getDefaultAlgorithm());
   587	        tmf.init((KeyStore) null);
   588	
   589	        TrustManager[] trustManagers = tmf.getTrustManagers();
   590	        final X509TrustManager origTrustManager = (X509TrustManager) trustManagers[0];
   591	
   592	        TrustManager[] wrappedTrustManagers = new TrustManager[]{
   593	                new X509TrustManager() {
   594	                    public java.security.cert.X509Certificate[] getAcceptedIssuers() {
   595	                        return origTrustManager.getAcceptedIssuers();
   596	                    }
   597	
   598	                    public void checkClientTrusted(X509Certificate[] certs, String authType) throws CertificateException {
   599	                        origTrustManager.checkClientTrusted(certs, authType);
   600	                    }
   601	
   602	                    public void checkServerTrusted(X509Certificate[] certs, String authType) throws CertificateException {
   603	                        origTrustManager.checkServerTrusted(certs, authType);
   604	                    }
   605	                }
   606	        };
   607	        SSLContext sslContext = SSLContext.getInstance("TLS");
   608	        sslContext.init(null, wrappedTrustManagers, null);
   609	
   610	        return sslContext;
   611	    }

   678	        try {
   679	            var info = appContext.getPackageManager().getPackageInfo(
   680	                    appContext.getPackageName(),
   681	                    PackageManager.GET_PERMISSIONS
   682	            );
   683	            if (info.requestedPermissions == null) {
   684	                return false;
   685	            }
   686	            for (var p : info.requestedPermissions) {
   687	                if ("android.permission.MANAGE_EXTERNAL_STORAGE".equals(p)) {
   688	                    return true;
   689	                }
   690	            }
   691	        } catch (PackageManager.NameNotFoundException e) {
   692	            Log.e(TAG, "Unable to get permissions", e);
   693	            return false;
   694	        }
   695	
   696	        return false;
   697	    }
