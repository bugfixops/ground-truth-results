   707	    private void initSearch() {

   718	                /* Submit the search will hide the keyboard */
   719	                searchView.clearFocus();
   720	
   721	                return true;
   722	            }
   723	
   724	            @Override
   725	            public boolean onQueryTextChange(String newText) {
   726	                viewModel.setSearchQuery(newText);

   731	        searchView.setQueryHint(getString(R.string.search));

   739	        int itemId = item.getItemId();
   740	        if (itemId == R.id.feed_menu) {
   741	            startActivity(new Intent(this, FeedActivity.class));
   742	        } else if (itemId == R.id.settings_menu) {
   743	            startActivity(new Intent(this, SettingsActivity.class));

   745	            showAboutDialog();
   746	        } else if (itemId == R.id.shutdown_app_menu) {
   747	            closeOptionsMenu();
   748	            viewModel.stopEngine();

   751	            viewModel.pauseAll();
   752	        } else if (itemId == R.id.resume_all_menu) {
   753	            viewModel.resumeAll();
   754	        } else if (itemId == R.id.log_menu) {
   755	            showLog();
   756	        }

   758	        return true;
