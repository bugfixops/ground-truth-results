    80	    private static boolean setStartStopAppAlarm(@NonNull Context appContext,
    81	                                                @NonNull String workTag, int time) {
    82	        Calendar calendar = Calendar.getInstance();
    83	        long timeInMillis = System.currentTimeMillis();
    84	
    85	        calendar.setTimeInMillis(timeInMillis);
    86	        calendar.set(Calendar.HOUR_OF_DAY, time / 60);
    87	        calendar.set(Calendar.MINUTE, time % 60);
    88	        calendar.set(Calendar.SECOND, 0);
    89	        calendar.set(Calendar.MILLISECOND, 0);
    90	        if (calendar.getTimeInMillis() < timeInMillis + 2000L)
    91	            calendar.add(Calendar.DAY_OF_MONTH, 1);
    92	
    93	        Intent intent = new Intent(appContext, SchedulerReceiver.class);
    94	        intent.setAction(workTag);
    95	        PendingIntent pi = PendingIntent.getBroadcast(
    96	                appContext, workTag.hashCode(), intent,
    97	                PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE
    98	        );
    99	        AlarmManager am = (AlarmManager) appContext.getSystemService(Context.ALARM_SERVICE);
   100	
   101	        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S && !am.canScheduleExactAlarms()) {
   102	            return false;
   103	        }
   104	        am.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, calendar.getTimeInMillis(), pi);
   105	        return true;
   106	    }

    41	public class SchedulerReceiver extends BroadcastReceiver
    42	{
    43	    @Override
    44	    public void onReceive(Context context, Intent intent)
    45	    {
    46	        if (intent.getAction() == null)
    47	            return;
    48	
    49	        var appContext = context.getApplicationContext();
    50	        var pref = RepositoryHelper.getSettingsRepository(appContext);
    51	        var notifier = TorrentNotifier.getInstance(appContext);
    52	
    53	        switch (intent.getAction()) {
    54	            case SCHEDULER_WORK_START_APP: {
    55	                onStartApp(appContext, pref, notifier);
    56	                break;
    57	            }
    58	            case SCHEDULER_WORK_STOP_APP: {
    59	                onStopApp(appContext, pref, notifier);
    60	                break;
    61	            }
    62	        }
    63	    }
    64	
    65	    private void onStartApp(Context appContext, SettingsRepository pref, TorrentNotifier notifier)
    66	    {
    67	        if (!pref.enableSchedulingStart())
    68	            return;
    69	
    70	        boolean oneshot = pref.schedulingRunOnlyOnce();
    71	        if (oneshot) {
    72	            pref.enableSchedulingStart(false);
    73	        } else if (!Scheduler.setStartAppAlarm(appContext, pref.schedulingStartTime())) {
    74	            notifier.makeExactAlarmPermissionNotify();
    75	        }
    76	        if (pref.schedulingSwitchWiFi())
    77	            ((WifiManager)appContext.getApplicationContext()
    78	                    .getSystemService(Context.WIFI_SERVICE)).setWifiEnabled(true);
    79	
    80	        TorrentEngine.getInstance(appContext).start();
    81	        Utils.enableBootReceiverIfNeeded(appContext);
    82	    }
    83	
    84	    private void onStopApp(Context appContext, SettingsRepository pref, TorrentNotifier notifier)
    85	    {
    86	        if (!pref.enableSchedulingShutdown())
    87	            return;
    88	
    89	        boolean oneshot = pref.schedulingRunOnlyOnce();
    90	        if (oneshot) {
    91	            pref.enableSchedulingShutdown(false);
    92	        } else if (!Scheduler.setStartAppAlarm(appContext, pref.schedulingShutdownTime())) {
    93	            notifier.makeExactAlarmPermissionNotify();
    94	        }
    95	
    96	        TorrentEngine.getInstance(appContext).forceStop();
    97	
    98	        if (pref.schedulingSwitchWiFi())
    99	            ((WifiManager)appContext.getApplicationContext()
   100	                    .getSystemService(Context.WIFI_SERVICE)).setWifiEnabled(false);
   101	
   102	        Utils.enableBootReceiverIfNeeded(appContext);
   103	    }
   104	}
