    20	package org.proninyaroslav.libretorrent.ui.addtorrent;
    21	
    22	import androidx.annotation.NonNull;
    23	import androidx.fragment.app.Fragment;
    24	import androidx.fragment.app.FragmentActivity;
    25	import androidx.viewpager2.adapter.FragmentStateAdapter;
    26	import androidx.viewpager2.widget.ViewPager2;
    27	
    28	public class AddTorrentPagerAdapter extends FragmentStateAdapter {
    29	    @ViewPager2.OffscreenPageLimit
    30	    public static final int NUM_FRAGMENTS = 2;
    31	
    32	    public static final int INFO_FRAG_POS = 0;
    33	    public static final int FILES_FRAG_POS = 1;
    34	
    35	    public AddTorrentPagerAdapter(FragmentActivity fa) {
    36	        super(fa);
    37	    }
    38	
    39	    @NonNull
    40	    @Override
    41	    public Fragment createFragment(int position) {
    42	        switch (position) {
    43	            case INFO_FRAG_POS:
    44	                return AddTorrentInfoFragment.newInstance();
    45	            case FILES_FRAG_POS:
    46	                return AddTorrentFilesFragment.newInstance();
    47	            default:
    48	                return new Fragment();
    49	        }
    50	    }
    51	
    52	    @Override
    53	    public int getItemCount() {
    54	        return NUM_FRAGMENTS;
    55	    }
    56	}
