   211	    private void subscribeAlertDialog() {
   212	        Disposable d = dialogViewModel.observeEvents()
   213	                .subscribe((event) -> {
   214	                    if (event.dialogTag == null || !event.dialogTag.equals(TAG_EDIT_NAME_DIALOG) || editNameDialog == null)
   215	                        return;
   216	                    switch (event.type) {
   217	                        case DIALOG_SHOWN:
   218	                            initEditNameDialog();
   219	                            break;
   220	                        case POSITIVE_BUTTON_CLICKED:
   221	                            Dialog dialog = editNameDialog.getDialog();
   222	                            if (dialog != null) {
   223	                                TextInputEditText editText = dialog.findViewById(R.id.text_input_dialog);
   224	                                TextInputLayout layoutEditText = dialog.findViewById(R.id.layout_text_input_dialog);
   225	                                Editable e = editText.getText();
   226	                                if (checkNameField(e, layoutEditText)) {
   227	                                    viewModel.mutableParams.setName(e.toString());
   228	                                    editNameDialog.dismiss();
   229	                                }
   230	                            }
   231	                            break;
   232	                        case NEGATIVE_BUTTON_CLICKED:
   233	                            editNameDialog.dismiss();
   234	                            break;
   235	                    }
   236	                });
   237	        disposables.add(d);
   238	    }

   259	    private boolean checkNameField(Editable s, TextInputLayout layoutEditText) {
   260	        if (TextUtils.isEmpty(s)) {
   261	            layoutEditText.setErrorEnabled(true);
   262	            layoutEditText.setError(getString(R.string.error_field_required));
   263	            layoutEditText.requestFocus();
   264	
   265	            return false;
   266	
   267	        } else {
   268	            layoutEditText.setErrorEnabled(false);
   269	            layoutEditText.setError(null);
   270	
   271	            return true;
   272	        }
   273	    }

   294	    private void initEditNameDialog() {
   295	        Dialog dialog = editNameDialog.getDialog();
   296	        if (dialog == null)
   297	            return;
   298	
   299	        TextInputEditText editText = dialog.findViewById(R.id.text_input_dialog);
   300	        TextInputLayout layoutEditText = dialog.findViewById(R.id.layout_text_input_dialog);
   301	
   302	        if (!TextUtils.isEmpty(editText.getText()))
   303	            return;
   304	
   305	        String name = viewModel.mutableParams.getName();
   306	        if (name != null)
   307	            editText.setText(name);
   308	
   309	        editText.addTextChangedListener(new TextWatcher() {
   310	            @Override
   311	            public void beforeTextChanged(CharSequence s, int start, int count, int after) {
   312	            }
   313	
   314	            @Override
   315	            public void onTextChanged(CharSequence s, int start, int before, int count) {
   316	            }
   317	
   318	            @Override
   319	            public void afterTextChanged(Editable s) {
   320	                checkNameField(s, layoutEditText);
   321	            }
   322	        });
   323	    }
