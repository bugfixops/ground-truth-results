353:         /* Use seed path parent; otherwise use save torrent file path */
354:         Uri savePath;
355:         Uri seedPath = mutableParams.getSeedPath().get();
356:         if (seedPath != null) {
357:             savePath = fs.getParentDirUri(seedPath);

361:             savePath = mutableParams.getSavePath();

365:             return Completable.complete();

373:                     .subscribeOn(Schedulers.io())
374:                     .subscribe((isRunning) -> {
375:                         if (isRunning) {

377:                             if (!emitter.isDisposed())
